def nan_corr(y_true, y_hat, alpha=0.05, mean_sub=True):
    """Ported verbatim from Creamer's metrics.py (bba43302), lines 109-136."""
    y_true = y_true.reshape(-1)
    y_hat = y_hat.reshape(-1)

    mask = ~np.isnan(y_true) & ~np.isnan(y_hat)
    y_true = y_true[mask]
    y_hat = y_hat[mask]

    if mean_sub:
        y_true = y_true - np.mean(y_true)
        y_hat = y_hat - np.mean(y_hat)

    y_true_std = np.std(y_true, ddof=0)
    y_hat_std = np.std(y_hat, ddof=0)

    corr = np.mean(y_true * y_hat) / y_true_std / y_hat_std

    # now estimate the confidence intervals for the correlation
    n = y_true.shape[0]
    z_a = scipy.stats.norm.ppf(1 - alpha / 2)
    z_r = np.log((1 + corr) / (1 - corr)) / 2
    l = z_r - (z_a / np.sqrt(n - 3))
    u = z_r + (z_a / np.sqrt(n - 3))
    ci_l = (np.exp(2 * l) - 1) / (np.exp(2 * l) + 1)
    ci_u = (np.exp(2 * u) - 1) / (np.exp(2 * u) + 1)
    ci = [np.abs(ci_l - corr), ci_u - corr]

    return corr, ci

def snapshot():
    e = top.encoder
    with torch.no_grad():
        return {"R": (e.chemical() + top.g_gap * e.gap_scale()).numpy().astype(np.float32),
                "inj": e.injection().numpy().astype(np.float32),
                "centres": None if e.pool_centres is None else e.pool_centres.detach().numpy().astype(np.float32),
                "readout": top.readout.detach().numpy().astype(np.float32),
                "bias": top.readout_bias.detach().numpy().astype(np.float32), "mode": e.readout_mode,
                "motor": top._motor_flat_indices.numpy(), "slices": list(top._motor_class_slices),
                "depth": top.forward_pass_depth}


def settle(S, h0, held=None):
    """Spike-local copy of the encoded forward: h <- tanh(h @ R + held), depth times; then readout."""
    h = np.asarray(h0, np.float32)
    for _ in range(S["depth"]):
        pre = h @ S["R"]
        if held is not None:
            pre = pre + np.asarray(held, np.float32)
        h = np.tanh(pre)
    acts = h[..., S["motor"]]
    pooled = np.stack([acts[..., a:b].mean(-1) for a, b in S["slices"]], -1)
    if S["centres"] is not None:
        pooled = pooled - S["centres"]
    mean = pooled @ S["readout"].T + S["bias"]
    if S["mode"] == "soft_cap":
        mean = cp.READOUT_CAP * np.tanh(mean / cp.READOUT_CAP)
    return mean, h


rng = np.random.default_rng(PARITY_SEED)
feats = np.column_stack([rng.uniform(0.05, 1.0, 100), rng.uniform(-1, 1, 100),
                         rng.uniform(-1, 1, 100), rng.uniform(0.05, 1.0, 100)]).astype(np.float32)
mods = rng.uniform(0.05, 1.0, (100, n_mod)).astype(np.float32)
idx = [top.neuron_names.index(n) for n in NAMES]
checks, responses, snaps = {}, {}, {}
for name, v in VECTORS.items():
    enc.write_encoder(brain, v)
    S = snaps[name] = snapshot()
    with torch.no_grad():
        m_pkg, h_pkg = top.forward_with_hidden_batched(torch.from_numpy(feats), modulators=torch.zeros(100, n_mod))
        m_mod, h_mod = top.forward_with_hidden_batched(torch.from_numpy(feats), modulators=torch.from_numpy(mods))
        held = top.modulator_current(torch.from_numpy(mods)).numpy()
        held_rest = top.modulator_current(torch.zeros(n_mod)).numpy()
    drive = feats @ S["inj"]
    m1, h1 = settle(S, drive)                       # parity, and unit test 1 (starting-state input = drive)
    m2, h2 = settle(S, drive, held=held)            # unit test 2 (held input = modulators' current)
    d1 = max(np.abs(m1 - m_pkg.numpy()).max(), np.abs(h1 - h_pkg.numpy()).max())
    d2 = max(np.abs(m2 - m_mod.numpy()).max(), np.abs(h2 - h_mod.numpy()).max())
    checks[name] = {"parity_and_unit_test_1_max_abs": float(d1), "unit_test_1_passes": bool(d1 <= TOL),
                    "unit_test_2_max_abs": float(d2), "unit_test_2_passes": bool(d2 <= TOL),
                    "unit_test_2_held_current_max_abs": float(np.abs(held).max()),
                    "sensory_drive_max_abs": float(np.abs(drive).max()),
                    "held_current_at_rest_max_abs": float(np.abs(held_rest).max()),
                    "rest_settled_max_abs": float(np.abs(settle(S, np.zeros(302, np.float32))[1]).max())}
    print(name, checks[name], flush=True)

def response(S, amp, form):
    """[responder, stimulated] over the 154, from rest: amp on neuron s's start (or held at every iteration)."""
    E = np.zeros((len(idx), 302), np.float32)
    E[np.arange(len(idx)), idx] = amp
    _, h = settle(S, E) if form == "start" else settle(S, np.zeros_like(E), held=E)
    return h[:, idx].T.astype(np.float64)       # rows: stimulated -> transpose to [responder, stimulated]

        for amp in (0.5, 1.0, 2.0):
            responses[(name, form, amp)] = response(snaps[name], amp, form)

def masked(M):
    M = M.copy(); M[eye] = np.nan; return M


def corr(M, ref):
    c, ci = nan_corr(masked(M), ref)
    return {"r": float(c), "ci95": [float(c - ci[0]), float(c + ci[1])]}


ceiling = corr(train, test)
ceiling["n_entries"] = int(np.sum(~np.isnan(train) & ~np.isnan(test)))
P = np.random.default_rng(PERM_SEED)
perms = [P.permutation(154) for _ in range(N_PERM)]


def null(M, ref):
    return np.array([nan_corr(masked(M[np.ix_(q, q)]), ref)[0] for q in perms])


def pct(x, dist):
    return float(np.mean(dist < x) * 100)


def sign_agreement(M, ref):
    M = masked(M); ok = ~np.isnan(ref) & ~np.isnan(M)
    mag = np.abs(ref[ok]); big = mag > np.median(mag)
    a, b = M[ok][big], ref[ok][big]
    return {"entries": int(big.sum()), "median_abs_measured": float(np.median(mag)),
            "agreement": float(np.mean(np.sign(a) == np.sign(b))), "model_zero_entries": int(np.sum(a == 0))}
