def nan_corr(y_true, y_hat, alpha=0.05, mean_sub=True):
    """Ported verbatim from Creamer's metrics.py (bba43302), lines 109-136."""
    y_true = y_true.reshape(-1)
    y_hat = y_hat.reshape(-1)

    mask = ~np.isnan(y_true) & ~np.isnan(y_hat)
    y_true = y_true[mask]
    y_hat = y_hat[mask]

    if mean_sub:
        y_true = y_true - np.mean(y_true)
        y_hat = y_hat - np.mean(y_hat)

    y_true_std = np.std(y_true, ddof=0)
    y_hat_std = np.std(y_hat, ddof=0)

    corr = np.mean(y_true * y_hat) / y_true_std / y_hat_std

    # now estimate the confidence intervals for the correlation
    n = y_true.shape[0]
    z_a = scipy.stats.norm.ppf(1 - alpha / 2)
    z_r = np.log((1 + corr) / (1 - corr)) / 2
    l = z_r - (z_a / np.sqrt(n - 3))
    u = z_r + (z_a / np.sqrt(n - 3))
    ci_l = (np.exp(2 * l) - 1) / (np.exp(2 * l) + 1)
    ci_u = (np.exp(2 * u) - 1) / (np.exp(2 * u) + 1)
    ci = [np.abs(ci_l - corr), ci_u - corr]

    return corr, ci


def settle_from_start(brain, h0):
    """M-a's spike-local settling copy, starting-state form: h <- tanh(h @ R), depth times."""
    top = brain.topology
    with torch.no_grad():
        R = (top.encoder.chemical() + top.g_gap * top.encoder.gap_scale()).numpy().astype(np.float32)
    h = np.asarray(h0, np.float32)
    for _ in range(top.forward_pass_depth):
        h = np.tanh(h @ R)
    return h


def response(brain, z):
    """[responder, stimulated] over the 154 names: one unit on neuron s's starting state."""
    enc.write_encoder(brain, z)
    names = brain.topology.neuron_names
    idx = [names.index(n) for n in NAMES]
    E = np.zeros((len(idx), len(names)), np.float32)
    E[np.arange(len(idx)), idx] = 1.0
    return settle_from_start(brain, E)[:, idx].T.astype(np.float64)


def edges(brain):
    """The built brain's chemical edges (directed, from its mask) and gap pairs (from its gap matrix)."""
    top = brain.topology
    chem = top.m_chem.to(torch.bool).numpy().copy()
    gap = (top.g_gap.detach().numpy() != 0)
    gap = np.triu(gap | gap.T, 1)
    return chem, gap


def package_forward_check(brain, z):
    """Parity of the settling copy against the package forward on this wiring (100 inputs, rng 9732)."""
    enc.write_encoder(brain, z)
    top = brain.topology
    rng = np.random.default_rng(9732)
    feats = np.column_stack([rng.uniform(0.05, 1.0, 100), rng.uniform(-1, 1, 100),
                             rng.uniform(-1, 1, 100), rng.uniform(0.05, 1.0, 100)]).astype(np.float32)
    with torch.no_grad():
        _, h_pkg = top.forward_with_hidden_batched(torch.from_numpy(feats), modulators=torch.zeros(100, brain._n_concentrations + 1))
        drive = (torch.from_numpy(feats) @ top.encoder.injection()).numpy()
    return float(np.abs(settle_from_start(brain, drive) - h_pkg.numpy()).max())
