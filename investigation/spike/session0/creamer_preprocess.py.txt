"""Session-0 re-implementation of Creamer's per-recording preparation, for fact-finding only.

Follows Creamer_LDS_2026 at bba43302 (MIT, Copyright (c) 2026 Matthew S. Creamer):
loading_utilities.preprocess_data, photobleach_correction and analysis_utilities.nan_convolve,
applied to a DANDI 001075 BaseGreenSignal (NaN frames kept, i.e. interpolate_nans False), with
stimulation inputs built from the NWB stimulation table: frame = first volume whose timestamp
+ STIM_DELTA_S >= start_time, target = target_pumpprobe_id (stimulations without one dropped),
amplitude 1. STIM_DELTA_S = 0.10065 s reproduced all 25 stimulation frames of his
example_recording.pkl (sub-4); it is not known to hold on other recordings.
"""

import warnings

import h5py
import numpy as np
import scipy.optimize as sopt

START_INDEX = 120
STIM_DELTA_S = 0.10065


def nan_convolve(data, filt):
    data = data.copy()
    nan_loc = np.isnan(data)
    data[nan_loc] = 0
    out = np.convolve(data, filt, mode="valid")
    cnt = np.convolve(~nan_loc, filt / filt.sum(), mode="valid")
    cnt[cnt == 0] = 1
    out = out / cnt
    pad = np.concatenate((nan_loc, np.zeros(filt.size - 1, bool)))[: out.shape[0]]
    out[pad] = np.nan
    return out


def photobleach(col, num_exp, fit_offset):
    x = col.copy()[:, None]
    if np.all(np.isnan(x)):
        return col.copy()
    t = np.arange(x.shape[0])
    tau_0 = t[-1, None] / np.arange(2 + num_exp, 2, -1)
    data_max = np.nanmax(x, axis=0)
    a_0 = np.concatenate([data_max / i for i in np.arange(2, 2 + num_exp)], axis=0)
    if fit_offset:
        p_0 = np.concatenate((np.log(tau_0), np.zeros(1), np.log(a_0)))
        amp_start = num_exp + 1
    else:
        p_0 = np.concatenate((np.log(tau_0), np.log(a_0)))
        amp_start = num_exp
    mask = ~np.isnan(x)
    x[~mask] = 0

    def approx(p):
        tau = np.exp(p[:num_exp])
        off = p[num_exp:amp_start]
        amp = np.exp(p[amp_start:])
        e = np.zeros_like(x)
        for k in range(num_exp):
            e = e + amp[k] * np.exp(-t[:, None] / tau[k])
        if fit_offset:
            e = e + off
        return e

    def loss(p):
        return (((approx(p) - x) ** 2) * mask).sum()

    p_hat = sopt.minimize(loss, p_0).x
    off = p_hat[num_exp:amp_start]
    e = approx(p_hat)
    corr = (x - off) / (e - off) if fit_offset else x / e
    corr[~mask] = np.nan
    return corr[:, 0]


def preprocess(em, inputs, start_index=START_INDEX):
    em = em[start_index:].copy()
    inputs = inputs[start_index:].copy()
    for c in range(em.shape[1]):
        for s0 in np.where(inputs[:, c])[0]:
            dx = np.array((-2, 3)) + s0
            ix = np.arange(dx[0], dx[1])
            if dx[1] < em.shape[0] and dx[0] >= 0:
                em[ix, c] = np.interp(ix, dx, em[dx, c])
    filt = np.ones(2) / 2
    emf = np.stack([nan_convolve(em[:, c], filt) for c in range(em.shape[1])], 1)
    out = np.zeros_like(emf)
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        for c in range(emf.shape[1]):
            out[:, c] = photobleach(emf[:, c], 2, False)
        bad2 = np.where(np.abs(np.nanmean(out, 0) - 1) > 0.1)[0]
        for c in bad2:
            out[:, c] = photobleach(emf[:, c], 1, True)
        bad1 = np.where(np.abs(np.nanmean(out, 0) - 1) > 0.2)[0]
        out[:, bad1] = np.nan
        out = out / np.nanmean(out, 0) - 1
    out[out > 5] = np.nan
    return out, inputs[: out.shape[0]], {"bad_2exp": bad2.tolist(), "bad_1exp": bad1.tolist()}


def load_recording(nwb_path, roi_labels, names):
    """Return (green[T, k], inputs[T, k], kept_names, stim_info) for the ROIs whose label is in
    names (first ROI per label, as align_data_cell_ids takes c.index(name))."""
    with h5py.File(nwb_path, "r") as f:
        oph = f["processing/ophys"]
        base = oph["GreenSignals/BaseGreenSignal/data"][()]
        ts = oph["GreenSignals/BaseGreenSignal/timestamps"][()]
        st = f["intervals/OptogeneticStimulusTable"]
        st_start = st["start_time"][()]
        tgt = st["target_pumpprobe_id"][()]
    name_set = set(names)
    cols, kept = [], []
    for j, lab in enumerate(roi_labels):
        if lab in name_set and lab not in kept:
            cols.append(j)
            kept.append(lab)
    inputs_roi = np.zeros(base.shape)
    frames = np.searchsorted(ts + STIM_DELTA_S, st_start)
    n_ident = 0
    for fr, m in zip(frames, tgt):
        if np.isnan(m) or fr >= base.shape[0]:
            continue
        inputs_roi[fr, int(m)] = 1
        n_ident += 1
    info = {"stims": len(st_start), "stims_identified": n_ident,
            "stims_on_kept": int(inputs_roi[:, cols].sum()),
            "stims_on_kept_after_cut": int(inputs_roi[START_INDEX:, cols].sum())}
    return base[:, cols], inputs_roi[:, cols], kept, info, ts
