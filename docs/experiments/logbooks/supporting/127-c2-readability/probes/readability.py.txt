"""C.2's readability probe: the guarded search on all six of v4's families (description, discarded seeds).

Logbook 126's guarded fit passes D_fit's four families, but fails every held-out family, and its
tph-1 and tax-4 mutants barely move: fitted on the lawn alone, nothing chooses which internal route
carries the worm's state. Before C.2's change registers anything, this probe asks two questions:

(a) can C.2 fit all six families, as C.1 is fitted (small lawn, uniform well, kinematics, activity,
    local search, chemotaxis)?
(b) once local search and chemotaxis are fitted, do the perturbations move, and which way:
    tph-1, pdfr-1 and tax-4 on the small lawn, graded on their v4 bands; and mgl-1 on López-Cruz's
    food removal, read beside López-Cruz's own same-date ratio?

The search is `guarded_wide.py`'s in every rule but its objective: `staged_bc.py`'s model (136
parameters, the soft cap, the pool centres, the graded box), the saturation guard (+500), the
responsiveness guard (+300; here over all four assays' recorded outputs) and sep-CMA at population
48, sigma0 0.1, from the guarded probe's answer (generation 250). The objective is v4's ranking
score against the band centres at n = 200 on every family. Presence is left out of it, as in C.1's
fits, and is graded at the checkpoints.

mgl-1 is encoded by removing the glutamatergic edges onto AIA and ADE that the López-Cruz sign
overrides sign inhibitory, the fast stand-in for MGL-1: the encoder's inhibitory entries that are
not inhibitory under `sign_override_exclude: [lopezcruz2019]`. Its reading is each 2-min bin's
reorientations per animal after removal, the early (8-12 min) and late (34-42 min) means and
mgl-1's ratio to its own wild type, on the same animals. Beside it is López-Cruz's ratio of mgl-1
to N2 recorded on the same dates (figures 3D and 3E).

Seeds: CMA 9479; fit animals 726,000,000 + 200·k + i with k = 48(g - 1) + m; checkpoints at
generations 0, 50, 100 and 150 on 727,000,000 + 200·c + i with c = g / 50; profiles on 99,670,000
as before; the smoke on 99,780,000. A cap of 150 generations on 16 workers, saved every generation
and resumable. Run from the repository root: `readability.py smoke`, then `readability.py`.
"""

import json
import pickle
import re

from pathlib import Path

import numpy as np
import torch

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c2-run-mode")
import benchmark_grade as bg

OUT = Path("build/c2-run-mode/readability")
START = Path("docs/experiments/logbooks/supporting/126-c2-run-mode/probes/guarded_wide.json")
LOPEZCRUZ = Path("data/benchmark/lopezcruz2019_reorientations.json")
POPSIZE, SIGMA0, CMA_SEED, CAP, EVERY = 48, 0.1, 9479, 150, 50
FIT_BASE, CHECK_BASE, SMOKE_BASE = 726_000_000, 727_000_000, 99_780_000
_MGL1_BRAIN = {}


def simulate(k, z, seed_base):
    """Run candidate k at z on all four assays; return its v4 statistics and both guards' inputs."""
    gw._OUTPUTS.clear()
    b._TALLY["steps"], b._TALLY["count"] = None, 0
    bands = bg.load_bands_v4()
    start = time.perf_counter()
    populations, chemotaxis = bg.simulate_v2(p.ARM, b.controller(z), p.population_seeds(seed_base, k), bands)
    graded = bg.grade_populations_v4(populations, chemotaxis, bands, presence=False)
    share = b._TALLY["steps"] / max(b._TALLY["count"], 1)
    out = np.concatenate(gw._OUTPUTS) if gw._OUTPUTS else np.zeros((1, 5))
    spread = np.percentile(out, 95, axis=0) - np.percentile(out, 5, axis=0)
    return nr._finite({
        "k": k, "z": list(map(float, z)),
        "statistics": {f"{f}/{n}": r["value"] for f, rs in graded["families"].items() for n, r in rs.items()},
        "z_scores": {f"{f}/{n}": r["z"] for f, rs in graded["families"].items() for n, r in rs.items()},
        "passes": {f: all(r["passes"] for r in rs.values()) for f, rs in graded["families"].items()},
        "D": graded["D"], "D_fit": graded["D_fit"],
        "motor_saturated_share": float(np.mean(share > b.SATURATED_SHARE)),
        "output_spread": dict(zip(("speed", "curvature", "reversal", "omega", "delta"), map(float, spread))),
        "seconds": time.perf_counter() - start,
    })


class Objective:
    """v4's ranking score on every family against the band centres at n = 200, with both guards."""

    def __init__(self):
        bands = bg.load_bands_v4()
        self.scales = v2.band_scales_v2(bands)
        self.family_map = {f: tuple(st) for f, st in bands["families"].items()}
        centres = {}
        for family, stats in self.family_map.items():
            family_bands = bg.family_bands_v2(bands, family, p.ANIMALS)
            for s in stats:
                centres[f"{family}/{s}"] = float(family_bands[s].centre)
        self.truth = {"statistics": centres}

    def __call__(self, record):
        score = v2.ranking_score(self.truth, record, self.scales, self.family_map)
        if record.get("motor_saturated_share", 0.0) > b.SATURATED_SHARE:
            score += b.PENALTY
        spread = record.get("output_spread", {})
        if min(spread.get("speed", 0.0), spread.get("reversal", 0.0)) < gw.SPREAD_FLOOR:
            score += gw.RESPONSIVE_PENALTY
        return score


def _mgl1_mask():
    """Return the inhibitory entries the López-Cruz overrides add, and their neuron pairs."""
    default = p.encoded_brain().topology.encoder.chem_negative
    excluded = p.encoded_brain((), sign_override_exclude=["lopezcruz2019"]).topology.encoder.chem_negative
    mask = (default != 0) & (excluded == 0)
    names = p.encoded_brain().topology.neuron_names
    pairs = [(names[int(i)], names[int(j)]) for i, j in zip(*np.nonzero(mask.numpy()), strict=True)]
    return mask, pairs


def mgl1_brain():
    """Return a fresh encoded brain with MGL-1's edges removed (built once per process)."""
    if "brain" not in _MGL1_BRAIN:
        mask, _ = _mgl1_mask()
        path = p.FORAGING / f"{bg.ARMS[p.ARM]}_assay_scheer_lawn.yml"
        settings = {**p._BRAIN_SETTINGS, "modulator_knockout_genes": [], "encoder_modulator_tying": b.TYING}
        brain, _, _ = p.harness.build_brain(path, p.BRAIN_SEED, settings)
        with torch.no_grad():
            brain.topology.encoder.chem_negative[mask] = 0.0
        _MGL1_BRAIN["brain"] = brain
    return _MGL1_BRAIN["brain"]


def controller(z, mutant=None):
    from quantumnematode.evaluation.controllers import BatchedConnectome

    brain = mgl1_brain() if mutant == "mgl-1" else p.encoded_brain(p.KNOCKOUT_GENES.get(mutant or "", ()))
    units = b.to_encoder(z)
    if mutant == "tax-4":
        units = list(enc.tax4_vector(brain.topology.modulator_spec, units, b.TYING))
    enc.write_encoder(brain, units)
    brain.topology.probe_centres = torch.from_numpy(-1.0 + 2.0 * np.asarray(z[b.N_ENCODER:], dtype=np.float32))
    return lambda _path: BatchedConnectome(brain)


def search_reading(animals):
    """Return a population's reorientation curve after removal, its early and late means, and its statistics."""
    onsets, valid = g._onsets(animals)
    curve = local_search_curve(onsets, valid, bin_s=g.SEARCH_BIN_S, start_s=0)
    by_start = {round(float(s) / 60.0, 6): float(m) for s, m in zip(curve.starts_s, curve.mean, strict=True)}
    early = float(np.mean([by_start[round(s, 6)] for s in g.EARLY_BINS_MIN]))
    late = float(np.mean([by_start[round(s, 6)] for s in g.LATE_BINS_MIN]))
    return {"curve": by_start, "early": early, "late": late, "statistics": g.simulated_search(animals)}


def lopezcruz_ratio():
    """Return López-Cruz's mgl-1 against same-date N2: early and late means, and their ratios."""
    deposit = json.loads(LOPEZCRUZ.read_text())
    recording = [int(re.match(r"(\d+)-", label).group(1)) for label in deposit["bins"]]
    # The deposit's bins are recording time, which began 4 min after removal.
    early = [recording.index(int(m) - 4) for m in g.EARLY_BINS_MIN]
    late = [recording.index(int(m) - 4) for m in g.LATE_BINS_MIN]
    out = {}
    for figure in ("3D", "3E"):
        rows = {r["genotype"]: r for r in deposit["rows"] if r["figure"] == figure and r["genotype"] in ("mgl-1", "N2")}
        e = {k: float(np.mean([r["mean"][i] for i in early])) for k, r in rows.items()}
        l_ = {k: float(np.mean([r["mean"][i] for i in late])) for k, r in rows.items()}
        out[figure] = {"early": e, "late": l_, "early_ratio": e["mgl-1"] / e["N2"], "late_ratio": l_["mgl-1"] / l_["N2"],
                       "n": {k: r["n"] for k, r in rows.items()}}
    return out


def wild(item):
    c, z = item
    bands = bg.load_bands_v4()
    seeds = p.population_seeds(CHECK_BASE, c)
    populations, chemotaxis = bg.simulate_v2(p.ARM, controller(z), seeds, bands)
    graded = bg.grade_populations_v4(populations, chemotaxis, bands, presence=True)
    clf = bg.classifier(bands)
    return c, bg._finite({
        "families": {f: {n: {"value": r["value"], "z": r["z"], "passes": r["passes"]} for n, r in rs.items()} for f, rs in graded["families"].items()},
        "family_passes": graded["family_passes"], "wild_type_passes": graded["wild_type_passes"],
        "local_search_presence": graded.get("local_search_presence"), "chemotaxis_presence": graded.get("chemotaxis_presence"),
        "D": graded["D"], "D_fit": graded["D_fit"], "reversal_share": graded["reversal_share"],
        "small_lawn_values": bg.family_values("small_lawn", populations["small_lawn"], clf),
        "search": search_reading(populations["local_search"]),
    })


def perturbed(item):
    c, name, z, wild_values = item
    bands = bg.load_bands_v4()
    seeds = p.population_seeds(CHECK_BASE, c)
    if name == "mgl-1":
        populations, _ = bg.simulate_v2(p.ARM, controller(z, name), seeds, bands, assays=("local_search",))
        return c, name, bg._finite(search_reading(populations["local_search"]))
    clf = bg.classifier(bands)
    populations, _ = bg.simulate_v2(p.ARM, controller(z, name), seeds, bands, assays=("small_lawn",))
    values = bg.family_values("small_lawn", populations["small_lawn"], clf)
    return c, name, p.mutant_reading(name, wild_values, values, bands, p.ANIMALS)


def checkpoint(pool, generation, z):
    c = generation // EVERY
    _, w = wild((c, z)) if pool is None else next(iter(pool.map(wild, [(c, z)])))
    jobs = [(c, m, z, w["small_lawn_values"]) for m in (*p.MUTANTS, "mgl-1")]
    for _, name, reading in (map(perturbed, jobs) if pool is None else pool.map(perturbed, jobs)):
        w.setdefault("perturbations", {})[name] = reading
    m = w["perturbations"]["mgl-1"]
    w["perturbations"]["mgl-1"] = {**m, "early_ratio": m["early"] / w["search"]["early"] if w["search"]["early"] else None,
                                   "late_ratio": m["late"] / w["search"]["late"] if w["search"]["late"] else None}
    w["generation"], w["profile"] = generation, b.profile(z)
    (OUT / f"check_{generation}.json").write_text(json.dumps(bg._finite(w), indent=1) + "\n")
    fails = [f"{f}/{n}" for f, rs in w["families"].items() for n, r in rs.items() if not r["passes"]]
    mut = {k: v.get("passes") for k, v in w["perturbations"].items() if k != "mgl-1"}
    print(f"check gen {generation}: D {w['D']:.2f} D_fit {w['D_fit']:.2f} families {w['family_passes']} fails {fails} "
          f"LS present {(w['local_search_presence'] or {}).get('present')} chem present {(w['chemotaxis_presence'] or {}).get('present')} "
          f"mutants {mut} mgl-1 early ratio {w['perturbations']['mgl-1']['early_ratio']} (wt early {w['search']['early']:.2f}) "
          f"profile {json.dumps(w['profile'])}", flush=True)


def start_vector():
    return list(json.loads(START.read_text())["answer"])


def smoke():
    _, pairs = _mgl1_mask()
    print("mgl-1 removes", len(pairs), "entries:", pairs, flush=True)
    print("López-Cruz", json.dumps(lopezcruz_ratio()), flush=True)
    record = simulate(0, start_vector(), SMOKE_BASE)
    print("smoke: score", round(Objective()(record), 3), "D", round(record["D"], 2), "D_fit", round(record["D_fit"], 2),
          "passes", record["passes"], "spread", {k: round(v, 2) for k, v in record["output_spread"].items()},
          "seconds", round(record["seconds"], 1), flush=True)


def main():
    import cma

    OUT.mkdir(parents=True, exist_ok=True)
    (OUT / "lopezcruz_ratio.json").write_text(json.dumps(lopezcruz_ratio(), indent=1) + "\n")
    state_path, history_path = OUT / "state.pkl", OUT / "history.json"
    if state_path.exists():
        es = pickle.loads(state_path.read_bytes())
        history = json.loads(history_path.read_text())
    else:
        es = cma.CMAEvolutionStrategy(start_vector(), SIGMA0, {"CMA_diagonal": True, "bounds": [0.0, 1.0], "popsize": POPSIZE, "seed": CMA_SEED, "verbose": -9})
        history = {"medians": [], "sigmas": [], "answers": [], "responsive_share": [], "best": [], "best_D": []}
    objective = Objective()
    t0 = time.time()
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        if not (OUT / "check_0.json").exists():
            checkpoint(pool, 0, start_vector())
        while len(history["medians"]) < CAP:
            gen = len(history["medians"]) + 1
            points = es.ask()
            ks = [POPSIZE * (gen - 1) + m for m in range(POPSIZE)]
            records = list(pool.map(simulate, ks, [list(map(float, x)) for x in points], [FIT_BASE] * POPSIZE))
            scores = [objective(r) for r in records]
            es.tell(points, scores)
            history["medians"].append(float(np.median(scores)))
            history["best"].append(float(np.min(scores)))
            history["best_D"].append(float(records[int(np.argmin(scores))]["D"]))
            history["sigmas"].append(float(es.sigma))
            history["answers"].append(np.clip(es.result.xfavorite, 0, 1).tolist())
            history["responsive_share"].append(float(np.mean([min(r["output_spread"]["speed"], r["output_spread"]["reversal"]) >= gw.SPREAD_FLOOR for r in records])))
            state_path.write_bytes(pickle.dumps(es))
            history_path.write_text(json.dumps(history))
            print(f"gen {gen}: median {history['medians'][-1]:.2f}, best {history['best'][-1]:.2f} (D {history['best_D'][-1]:.2f}), "
                  f"responsive {history['responsive_share'][-1]:.2f}, sigma {es.sigma:.3f}, {time.time() - t0:.0f} s", flush=True)
            if gen % EVERY == 0 and not (OUT / f"check_{gen}.json").exists():
                checkpoint(pool, gen, history["answers"][gen - 1])
    checks = {int(f.stem.split("_")[1]): json.loads(f.read_text()) for f in sorted(OUT.glob("check_*.json"))}
    record = {"description": __doc__, "popsize": POPSIZE, "cap": CAP, **history, "answer": history["answers"][-1],
              "checkpoints": checks, "lopezcruz_ratio": lopezcruz_ratio(), "mgl1_removed": _mgl1_mask()[1],
              "wall_hours": (time.time() - t0) / 3600}
    (OUT.parent / "readability.json").write_text(json.dumps(nr._finite(record), indent=1) + "\n")


if __name__ == "__main__":
    smoke() if sys.argv[1:] == ["smoke"] else main()
