"""C.1's 25-parameter model on v3: existence and basin frequency (description, discarded seeds).

warm:  x0 = the 24-parameter speed-spread answer, with kappa_off_food at a quarter of its on-food
       gain; sigma0 0.1; CMA 5766; fit 89,000,000; graded on 89,970,000.
boxA:  the box centre, sigma0 0.2, CMA 5767; fit 88,000,000; graded on 88,970,000.
boxB:  the box centre, sigma0 0.2, CMA 5768; fit 87,000,000; graded on 87,970,000.
All: v3's ranking score against the band centres at n = 200, C.1b's guarded stop, cap 300; the
answer graded with presence on 5 populations. Reaches: at least 3 of 5 pass.
"""

import json
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c1-v3")
import benchmark_grade as bg
import benchmark_v3_controls as v3c
import c1b_fit as c1b
import cma_recovery as cr
import full25_probe as f25
import npe_recovery as nr
import readiness_probe as rp

RUNS = {
    "warm": {"cma": 5766, "fit": 89_000_000, "grade": 89_970_000, "sigma0": 0.1},
    "boxA": {"cma": 5767, "fit": 88_000_000, "grade": 88_970_000, "sigma0": 0.2},
    "boxB": {"cma": 5768, "fit": 87_000_000, "grade": 87_970_000, "sigma0": 0.2},
}


def warm_x0():
    spread = json.loads(Path("build/c1-v3/probe_spread.json").read_text())
    kappa = spread["parameters"]["kappa_weathervane"] * 0.25
    low, high = f25.EXTRA["kappa_off_food"]
    return [*spread["answer"], (kappa - low) / (high - low)]


def start_warm(seed):
    import cma

    return cma.CMAEvolutionStrategy(warm_x0(), RUNS["warm"]["sigma0"], {"CMA_diagonal": True, "bounds": [0.0, 1.0], "popsize": cr.POPSIZE, "seed": seed, "verbose": -9})


def grade(item):
    z, base, gpop = item
    graded = v3c._mechanistic_graded(f25.parameters(z), v3c.population_seeds(base, gpop), bg.load_bands_v3())
    return bg._finite({
        "g": gpop, "wild_type_passes": graded["wild_type_passes"], "family_passes": graded["family_passes"],
        "D": graded["D"], "D_fit": graded["D_fit"],
        "failing": [f"{f}/{n}" for f, rs in graded["families"].items() for n, r in rs.items() if not r["passes"]],
        "z": {f"{f}/{n}": r["z"] for f, rs in graded["families"].items() for n, r in rs.items()},
        "chemotaxis_present": graded["chemotaxis_presence"]["present"],
        "local_search_present": graded["local_search_presence"]["present"],
        "reversal_share": graded["reversal_share"]["value"] if graded["reversal_share"] else None,
    })


def run(name, pool):
    p = RUNS[name]
    out = Path(f"build/c1-v3/basin_{name}.json")
    if out.exists():
        return
    t0 = time.time()
    history = cr.run_fit(
        {}, fit=0, cma_seed=p["cma"], seed_base=p["fit"], cap=300, use_rule=True, state_dir=Path(f"build/c1-v3/basin_{name}"),
        pool=pool, scales={}, objective=rp.V3BandObjective(), simulate=f25.simulate, stop=c1b.defined_stop,
        strategy_factory=start_warm if name == "warm" else f25.start,
    )
    answer = history["answers"][-1]
    grades = list(pool.map(grade, [(answer, p["grade"], g) for g in range(5)]))
    passing = sum(bool(g["wild_type_passes"]) for g in grades)
    record = {
        "description": f"Description: C.1's 25-parameter model on v3, run '{name}'; discarded seeds.",
        **p, "stop": history.get("stop"), "generations": len(history["medians"]), "medians": history["medians"],
        "sigmas": history["sigmas"], "answer": answer, "parameters": f25.parameters(answer),
        "populations_passing": passing, "reaches": passing >= 3, "grades": grades, "wall_hours": (time.time() - t0) / 3600,
    }
    out.write_text(json.dumps(bg._finite(record), indent=2) + "\n")
    print(name, json.dumps({k: record[k] for k in ("stop", "generations", "populations_passing", "reaches")}), "D", [round(g["D"], 2) for g in grades], flush=True)
    for g in grades:
        print("  ", g["g"], g["wild_type_passes"], g["failing"], flush=True)


def main():
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        for name in RUNS:
            run(name, pool)


if __name__ == "__main__":
    main()
