"""C.1 on v3: two diagnostic probes after the readiness probe failed (description, discarded seeds).

(a) warm: C.1b's chosen answer (restart 2) as x0, sigma0 0.1, C.1's box, CMA 5761, fit animals at
    97,000,000, graded on 97,970,000.
(b) wide: the box centre at sigma0 0.2, with the bounds the readiness probe hit widened (sigma_dwell
    to 180, sigma_roam from 0.5, search_tau_s from 1 s, a_intake to 10, beta_omega to 10), CMA
    5762, fit animals at 96,000,000, graded on 96,970,000.
Both: v3's ranking score against the band centres at n = 200, C.1b's guarded stop, cap 300, and
the answer graded on v3 with presence on 5 populations. Reaches: at least 3 of 5 pass.
"""

import json
import math
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c1-v3")
import benchmark_grade as bg
import benchmark_v3_controls as v3c
import c1b_fit as c1b
import cma_recovery as cr
import npe_recovery as nr
import readiness_probe as rp

WIDE = {
    "sigma_dwell": ("uniform", 5.0, 180.0),
    "sigma_roam": ("uniform", 0.5, 60.0),
    "search_tau_s": ("log", 1.0, 3000.0),
    "a_intake": ("uniform", -5.0, 10.0),
    "beta_omega": ("uniform", 0.0, 10.0),
}
PROBES = {
    "warm": {"cma": 5761, "fit": 97_000_000, "grade": 97_970_000, "sigma0": 0.1},
    "wide": {"cma": 5762, "fit": 96_000_000, "grade": 96_970_000, "sigma0": 0.2},
}


def wide_parameters(z):
    from quantumnematode.evaluation.controllers import MECHANISTIC_PARAMETERS

    out = {}
    for value, name in zip(np.asarray(z, dtype=float), MECHANISTIC_PARAMETERS, strict=True):
        kind, low, high = WIDE.get(name, nr.PRIOR[name])
        out[name] = float(math.exp(math.log(low) + value * (math.log(high) - math.log(low)))) if kind == "log" else float(low + value * (high - low))
    return out


def _record(k, z, seed_base, params):
    from quantumnematode.evaluation.controllers import BatchedMechanistic

    bands = bg.load_bands_v3()
    seeds = [seed_base + nr.ANIMALS * k + i for i in range(nr.ANIMALS)]
    start = time.perf_counter()
    populations, chemotaxis = bg.simulate_v2("mlp", lambda _p: BatchedMechanistic(params), seeds, bands)
    graded = bg.grade_populations_v3(populations, chemotaxis, bands, presence=False)
    statistics = {f"{f}/{n}": r["value"] for f, rs in graded["families"].items() for n, r in rs.items()}
    return nr._finite({"k": k, "z": list(map(float, z)), "statistics": statistics, "D": graded["D"], "D_fit": graded["D_fit"], "seconds": time.perf_counter() - start})


def simulate_warm(k, z, seed_base):
    return _record(k, z, seed_base, nr.parameters_from_unit(np.asarray(z)))


def simulate_wide(k, z, seed_base):
    return _record(k, z, seed_base, wide_parameters(z))


def grade(item):
    name, z, gpop = item
    params = wide_parameters(z) if name == "wide" else nr.parameters_from_unit(np.asarray(z))
    graded = v3c._mechanistic_graded(params, v3c.population_seeds(PROBES[name]["grade"], gpop), bg.load_bands_v3())
    return bg._finite({
        "g": gpop, "wild_type_passes": graded["wild_type_passes"], "family_passes": graded["family_passes"],
        "D": graded["D"], "D_fit": graded["D_fit"],
        "failing": [f"{f}/{n}" for f, rs in graded["families"].items() for n, r in rs.items() if not r["passes"]],
        "z": {f"{f}/{n}": r["z"] for f, rs in graded["families"].items() for n, r in rs.items()},
        "chemotaxis_present": graded["chemotaxis_presence"]["present"],
        "local_search_present": graded["local_search_presence"]["present"],
        "reversal_share": graded["reversal_share"]["value"] if graded["reversal_share"] else None,
    })


def warm_x0():
    record = json.loads(c1b.FIT_PATH.read_text())
    return record["restarts"][record["chosen_restart"]]["answer"]


def start_warm(seed):
    import cma

    return cma.CMAEvolutionStrategy(warm_x0(), PROBES["warm"]["sigma0"], {"CMA_diagonal": True, "bounds": [0.0, 1.0], "popsize": cr.POPSIZE, "seed": seed, "verbose": -9})


def run(name, pool):
    p = PROBES[name]
    t0 = time.time()
    history = cr.run_fit(
        {}, fit=0, cma_seed=p["cma"], seed_base=p["fit"], cap=300, use_rule=True, state_dir=Path(f"build/c1-v3/{name}"),
        pool=pool, scales={}, objective=rp.V3BandObjective(), simulate=simulate_warm if name == "warm" else simulate_wide,
        stop=c1b.defined_stop, strategy_factory=start_warm if name == "warm" else None,
    )
    answer = history["answers"][-1]
    grades = list(pool.map(grade, [(name, answer, g) for g in range(5)]))
    passing = sum(bool(g["wild_type_passes"]) for g in grades)
    params = wide_parameters(answer) if name == "wide" else nr.parameters_from_unit(np.asarray(answer))
    record = {
        "description": f"Description: C.1 diagnostic probe '{name}' on v3, discarded seeds.",
        **p, "stop": history.get("stop"), "generations": len(history["medians"]),
        "medians": history["medians"], "sigmas": history["sigmas"], "answer": answer, "parameters": params,
        "populations_passing": passing, "reaches": passing >= 3, "grades": grades, "wall_hours": (time.time() - t0) / 3600,
    }
    Path(f"build/c1-v3/probe_{name}.json").write_text(json.dumps(bg._finite(record), indent=2) + "\n")
    print(name, json.dumps({k: record[k] for k in ("stop", "generations", "populations_passing", "reaches")}), flush=True)
    for g in grades:
        print(" ", g["g"], g["wild_type_passes"], round(g["D"], 2) if g["D"] else None, g["failing"], "chem", g["chemotaxis_present"], "rev", round(g["reversal_share"], 3) if g["reversal_share"] else None, flush=True)


def main():
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        for name in sys.argv[1:] or ("warm", "wide"):
            run(name, pool)


if __name__ == "__main__":
    main()
