"""Readiness probe for C.1 on benchmark v3 (description, discarded seeds, before any change).

One sep-CMA restart of C.1's 20-parameter model from the box centre, sigma0 0.2, population 16,
CMA seed 5760, against v3's 21 graded statistics: C.1b's ranking score (definedness first, then a
compressed distance) against v3's band centres and scales at n = 200. C.1b's stop rule: the plateau
rule once sigma <= sigma0 and its last 40 generations hold no undefined median, or the cap of 300.
Fit animals 98,000,000 + 200·k + i (k = 16·(n - 1) + m). The answer is graded on v3 with presence
on 5 populations at 98,970,000 + 200·g + i. Ready: at least 3 of 5 pass the wild type.
"""

import json
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
import benchmark_grade as bg
import benchmark_v3_controls as v3c
import c1b_fit as c1b
import cma_recovery as cr
import cma_recovery_v2 as v2
import npe_recovery as nr

CMA_SEED = 5760
FIT_BASE = 98_000_000
GRADE_BASE = 98_970_000
CAP = 300
STATE = Path("build/c1-v3/probe")
OUT = Path("build/c1-v3/readiness_probe.json")


class V3BandObjective:
    """The ranking score of a v3 simulation against v3's band centres, picklable."""

    def __init__(self) -> None:
        bands = bg.load_bands_v3()
        self.family_map = {f: tuple(st) for f, st in bands["families"].items()}
        self.scales = v2.band_scales_v2(bands)
        centres = {}
        for family, stats in self.family_map.items():
            family_bands = bg.family_bands_v2(bands, family, nr.ANIMALS)
            for s in stats:
                centres[f"{family}/{s}"] = float(family_bands[s].centre)
        self.truth = {"statistics": centres}

    def __call__(self, record):
        return v2.ranking_score(self.truth, record, self.scales, self.family_map)


def simulate_v3(k, z, seed_base):
    """Run simulation k of C.1's model at z on v3's four assays; return its graded statistics."""
    from quantumnematode.evaluation.controllers import BatchedMechanistic

    bands = bg.load_bands_v3()
    seeds = [seed_base + nr.ANIMALS * k + i for i in range(nr.ANIMALS)]
    params = nr.parameters_from_unit(np.asarray(z))
    start = time.perf_counter()
    populations, chemotaxis = bg.simulate_v2("mlp", lambda _p: BatchedMechanistic(params), seeds, bands)
    graded = bg.grade_populations_v3(populations, chemotaxis, bands, presence=False)
    statistics = {f"{f}/{n}": r["value"] for f, rs in graded["families"].items() for n, r in rs.items()}
    share = graded["reversal_share"]
    record = {
        "k": k,
        "z": list(map(float, z)),
        "statistics": statistics,
        "D": graded["D"],
        "D_fit": graded["D_fit"],
        "reversal_share": None if share is None else share["value"],
        "seconds": time.perf_counter() - start,
    }
    return nr._finite(record)


def grade(g):
    z, gpop = g
    bands = bg.load_bands_v3()
    graded = v3c._mechanistic_graded(nr.parameters_from_unit(np.asarray(z)), v3c.population_seeds(GRADE_BASE, gpop), bands)
    failing = [f"{f}/{n}" for f, rs in graded["families"].items() for n, r in rs.items() if not r["passes"]]
    return bg._finite({
        "g": gpop,
        "wild_type_passes": graded["wild_type_passes"],
        "family_passes": graded["family_passes"],
        "D": graded["D"],
        "D_fit": graded["D_fit"],
        "failing": failing,
        "z": {f"{f}/{n}": r["z"] for f, rs in graded["families"].items() for n, r in rs.items()},
        "chemotaxis_present": graded["chemotaxis_presence"]["present"],
        "local_search_present": graded["local_search_presence"]["present"],
        "reversal_share": graded["reversal_share"]["value"] if graded["reversal_share"] else None,
    })


def main():
    t0 = time.time()
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        history = cr.run_fit(
            {}, fit=0, cma_seed=CMA_SEED, seed_base=FIT_BASE, cap=CAP, use_rule=True, state_dir=STATE,
            pool=pool, scales={}, objective=V3BandObjective(), simulate=simulate_v3, stop=c1b.defined_stop,
        )
        answer = history["answers"][-1]
        grades = list(pool.map(grade, [(answer, g) for g in range(5)]))
    passing = sum(bool(g["wild_type_passes"]) for g in grades)
    record = {
        "description": "Description: C.1 readiness probe on v3, discarded seeds, before any change.",
        "cma_seed": CMA_SEED, "fit_base": FIT_BASE, "grade_base": GRADE_BASE, "cap": CAP,
        "stop": history.get("stop"), "generations": len(history["medians"]),
        "plateau_generation": c1b.defined_plateau_generation(history["medians"], history["sigmas"]),
        "medians": history["medians"], "sigmas": history["sigmas"],
        "seconds_per_generation": float(np.median(history["seconds"])) if history.get("seconds") else None,
        "answer": answer, "parameters": nr.parameters_from_unit(np.asarray(answer)),
        "populations_passing": passing, "ready": passing >= 3, "grades": grades,
        "wall_hours": (time.time() - t0) / 3600,
    }
    OUT.write_text(json.dumps(bg._finite(record), indent=2) + "\n")
    print(json.dumps({k: record[k] for k in ("stop", "generations", "plateau_generation", "populations_passing", "ready")}))
    for g in grades:
        print(g["g"], g["wild_type_passes"], round(g["D"], 2) if g["D"] else None, g["failing"], "chem present", g["chemotaxis_present"], "rev", g["reversal_share"])


if __name__ == "__main__":
    main()
