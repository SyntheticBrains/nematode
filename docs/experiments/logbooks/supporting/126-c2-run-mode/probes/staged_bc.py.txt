"""C.2's staged fit redesigned (description, discarded seeds): a stage 1 that switches the network on,
and fewer parameters (per-ligand tying), on benchmark v4's D_fit.

The plan review (`build/c2-fit-review-2026-11.md`, options B and C): stage 1 of the earlier staged
fit never fitted the amphid sensory gains, so its network stayed silent, and stage 2's 196
dimensions found bias-driven optima. Here, applied in this script (no package code changes):

- the model: the encoded connectome under per-ligand modulator tying (132 parameters), with four
  fitted pool centres (136);
- the box: the three wiring scales on [0, 0.75] and every sensory gain on [-2, 2], each mapped onto
  the encoder's unit box; every other range unchanged;
- the readout: each output's pre-squash value through 2·tanh(x / 2), read as b + Σ_j w_j (pool_j - c_j);
- the guard: a candidate whose motor layer saturates (over half its motor neurons at |h| > 0.99 in
  over half the steps) scores 500 more;
- stage 1: the 77 parameters that carry the senses to the outputs (the 7 wiring terms, the 36
  sensory gains, the 25 readout weights and biases, the 5 noise terms, the 4 centres), every other
  at the box centre; from the box centre, sigma0 0.2; CMA 9474; fit animals 708,000,000; the guarded
  readiness stop or a cap of 200;
- stage 2: all 136 from stage 1's answer, sigma0 0.1; CMA 9475; fit animals 710,000,000; run to 200.

Both stages: v4's ranking score on D_fit (with the guard), on the corrected wiring, 16 workers.
Every 50 generations, and at each stage's end, the answer is profiled (bounded pre-squash
percentiles, primitive shares, time on the lawn, motor saturation, each food feature's gain onto
the pools; seed 99,670,000) and graded on D_fit's assays on one population (stage 1 at
709,000,000, stage 2 at 711,000,000). Run from the repository root.
"""

import json
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np
import torch

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c2-run-mode")
import benchmark_grade as bg
import c2_encoder_probe as p
import cma_recovery as cr
import npe_recovery as nr
import stage2_v4 as s
from quantumnematode.brain.arch import connectome_ppo as cp
from quantumnematode.connectome import encoder as enc

TYING = "per_ligand"
p._BRAIN_SETTINGS["encoder_modulator_tying"] = TYING
OUT = Path("build/c2-run-mode")
SOFT_CAP, SATURATED, SATURATED_SHARE, PENALTY = 2.0, 0.99, 0.5, 500.0
WIRING_HIGH, SENSORY_BOUND = 0.75, 2.0
POOLS = 4
STAGES = {
    1: {"cma": 9474, "fit": 708_000_000, "check": 709_000_000, "sigma0": 0.2, "cap": 200, "rule": True},
    2: {"cma": 9475, "fit": 710_000_000, "check": 711_000_000, "sigma0": 0.1, "cap": 200, "rule": False},
}
EVERY = 50
STAGE1_BLOCKS = ("wiring", "sensory", "readout", "noise")
_TALLY = {"steps": None, "count": 0}


def _forward(self, h, modulators):
    encoder = self.encoder
    recurrent = encoder.chemical() + self.g_gap * encoder.gap_scale()
    held = self._held_current(modulators)
    previous = h
    for _ in range(self.forward_pass_depth):
        previous = h
        preact = h @ recurrent
        if held is not None:
            preact = preact + held
        h = cp._settling_tanh(preact)
    if self.capture_penultimate:
        self.penultimate = previous.detach()
    motor = h.detach().index_select(-1, self._motor_flat_indices).abs() > SATURATED
    motor = motor.reshape(-1, motor.shape[-1]).sum(dim=0).numpy()
    _TALLY["steps"] = motor if _TALLY["steps"] is None else _TALLY["steps"] + motor
    _TALLY["count"] += int(h.reshape(-1, h.shape[-1]).shape[0])
    pooled = self._pool_motor(h)
    centres = getattr(self, "probe_centres", None)
    if centres is not None:
        pooled = pooled - centres
    mean = pooled @ self.readout.T + self.readout_bias
    return SOFT_CAP * torch.tanh(mean / SOFT_CAP), h


cp.ConnectomeTopology._forward_encoded = _forward


def entries():
    return enc.encoder_entries(p.layer_spec(), TYING)


N_ENCODER = len(entries())
STAGE1_INDEX = [i for i, e in enumerate(entries()) if e.block in STAGE1_BLOCKS] + list(range(N_ENCODER, N_ENCODER + POOLS))


def to_encoder(z):
    out = list(map(float, z[:N_ENCODER]))
    for i, e in enumerate(entries()):
        if e.name in ("wiring.excitatory", "wiring.inhibitory", "wiring.gap"):
            out[i] = z[i] * WIRING_HIGH / 1.5
        elif e.name.startswith("sensory."):
            out[i] = (-SENSORY_BOUND + 2 * SENSORY_BOUND * z[i] + 5.0) / 10.0
    return out


def write(brain, z):
    enc.write_encoder(brain, to_encoder(z))
    brain.topology.probe_centres = torch.from_numpy(-1.0 + 2.0 * np.asarray(z[N_ENCODER:], dtype=np.float32))


def controller(z):
    from quantumnematode.evaluation.controllers import BatchedConnectome

    brain = p.encoded_brain()
    write(brain, z)
    return lambda _path: BatchedConnectome(brain)


def _simulate_full(k, z, seed_base):
    bands = bg.load_bands_v4()
    _TALLY["steps"], _TALLY["count"] = None, 0
    start = time.perf_counter()
    populations, _ = bg.simulate_v2(p.ARM, controller(z), p.population_seeds(seed_base, k), bands, assays=p.FIT_ASSAYS)
    graded = bg.grade_populations_v4(populations, None, bands, presence=False)
    share = _TALLY["steps"] / max(_TALLY["count"], 1)
    return nr._finite({
        "k": k, "z": list(map(float, z)),
        "statistics": {f"{f}/{n}": r["value"] for f, rs in graded["families"].items() for n, r in rs.items()},
        "z_scores": {f"{f}/{n}": r["z"] for f, rs in graded["families"].items() for n, r in rs.items()},
        "fit_passes": {f: all(r["passes"] for r in graded["families"][f].values()) for f in bands["fit_families"]},
        "motor_saturated_share": float(np.mean(share > SATURATED_SHARE)),
        "D_fit": graded["D_fit"], "seconds": time.perf_counter() - start,
    })


def embed(x):
    """Return the full 136-vector: stage 1's values in place, every other at the box centre."""
    z = [0.5] * (N_ENCODER + POOLS)
    for i, v in zip(STAGE1_INDEX, x, strict=True):
        z[i] = float(v)
    return z


def simulate_stage1(k, x, seed_base):
    return {**_simulate_full(k, embed(x), seed_base), "x": list(map(float, x))}


def simulate_stage2(k, z, seed_base):
    return _simulate_full(k, z, seed_base)


class GuardedObjective(s.V4FitObjective):
    def __call__(self, record):
        return super().__call__(record) + (PENALTY if record.get("motor_saturated_share", 0.0) > SATURATED_SHARE else 0.0)


def _strategy(x0, sigma0, seed):
    import cma

    return cma.CMAEvolutionStrategy(x0, sigma0, {"CMA_diagonal": True, "bounds": [0.0, 1.0], "popsize": cr.POPSIZE, "seed": seed, "verbose": -9})


def start_stage1(seed):
    return _strategy([0.5] * len(STAGE1_INDEX), STAGES[1]["sigma0"], seed)


def start_stage2(seed):
    x0 = json.loads((OUT / "staged_bc" / "stage1_answer.json").read_text())["z"]
    return _strategy(x0, STAGES[2]["sigma0"], seed)


def profile(z):
    from quantumnematode.evaluation.reference import run_reference
    from quantumnematode.utils.config_loader import load_simulation_config

    path = p.FORAGING / f"{bg.ARMS[p.ARM]}_assay_scheer_lawn.yml"
    brain = p.encoded_brain()
    write(brain, z)
    episode = run_reference(load_simulation_config(str(path)), brain, s.PROFILE_SEED, 1800, record_density=False)
    prim = np.array([str(x.primitive) for x in episode.behaviour])
    on = np.array([bool(x.on_lawn) for x in episode.behaviour])
    states = torch.from_numpy(np.asarray(episode.features, dtype=np.float32))
    topo = brain.topology
    nf = topo.n_food_features
    mods = brain._unpack_modulators(states)
    with torch.no_grad():
        mean, hidden = topo.forward_with_hidden_batched(states[:, :nf], modulators=mods)
        idx = torch.from_numpy(np.random.default_rng(0).choice(len(states), size=min(200, len(states)), replace=False))
        gains = []
        for j in range(nf):
            up, down = states[idx, :nf].clone(), states[idx, :nf].clone()
            up[:, j] += 0.05
            down[:, j] -= 0.05
            _, hu = topo.forward_with_hidden_batched(up, modulators=mods[idx])
            _, hd = topo.forward_with_hidden_batched(down, modulators=mods[idx])
            gains.append(float(np.abs((topo._pool_motor_classes(hu) - topo._pool_motor_classes(hd)).numpy() / 0.1).mean()))
    motor = np.abs(hidden.numpy()[:, topo._motor_flat_indices.numpy()]) > SATURATED
    out = {o: np.percentile(mean[:, i].numpy(), [5, 50, 95]).round(2).tolist() for i, o in enumerate(("speed", "curvature", "reversal", "omega", "delta"))}
    out["primitive_shares"] = {str(k): round(float(np.mean(prim == k)), 3) for k in sorted(set(prim))}
    out["time_on_lawn"] = round(float(on.mean()), 3)
    out["motor_saturated_share"] = round(float(np.mean(motor.mean(axis=0) > SATURATED_SHARE)), 3)
    out["feature_to_pool_gain"] = [round(g, 4) for g in gains]
    return out


def _check(item):
    g, z, base = item
    return g, _simulate_full(g, z, base)


def _stage(stage, pool, history, to_full):
    gens = sorted(set(range(EVERY, len(history["answers"]) + 1, EVERY)) | {len(history["answers"])})
    checks = dict(pool.map(_check, [(g, to_full(history["answers"][g - 1]), STAGES[stage]["check"]) for g in gens]))
    points = []
    for g in gens:
        z = to_full(history["answers"][g - 1])
        c = checks[g]
        point = {"generation": g, "profile": profile(z), "D_fit": c["D_fit"], "fit_passes": c["fit_passes"],
                 "motor_saturated_share_sim": c["motor_saturated_share"], "z": c["z_scores"]}
        points.append(point)
        print(f"stage {stage} gen {g}", json.dumps(point["profile"]), "D_fit", round(c["D_fit"], 2), c["fit_passes"], flush=True)
    return {**STAGES[stage], "stop": history.get("stop"), "generations": len(history["medians"]), "medians": history["medians"],
            "sigmas": history["sigmas"], "answer": to_full(history["answers"][-1]), "points": points}


def main():
    t0 = time.time()
    record = {"description": __doc__, "n_parameters": N_ENCODER + POOLS, "stage1_parameters": len(STAGE1_INDEX)}
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        first = cr.run_fit({}, fit=0, cma_seed=STAGES[1]["cma"], seed_base=STAGES[1]["fit"], cap=STAGES[1]["cap"], use_rule=True,
                           state_dir=OUT / "staged_bc" / "stage1", pool=pool, scales={}, objective=GuardedObjective(),
                           simulate=simulate_stage1, stop=p.readiness_stop, strategy_factory=start_stage1)
        (OUT / "staged_bc").mkdir(parents=True, exist_ok=True)
        (OUT / "staged_bc" / "stage1_answer.json").write_text(json.dumps({"z": embed(first["answers"][-1])}))
        record["stage1"] = _stage(1, pool, first, embed)
        second = cr.run_fit({}, fit=0, cma_seed=STAGES[2]["cma"], seed_base=STAGES[2]["fit"], cap=STAGES[2]["cap"], use_rule=False,
                            state_dir=OUT / "staged_bc" / "stage2", pool=pool, scales={}, objective=GuardedObjective(),
                            simulate=simulate_stage2, stop=p.readiness_stop, strategy_factory=start_stage2)
        record["stage2"] = _stage(2, pool, second, lambda z: list(z))
    record["wall_hours"] = (time.time() - t0) / 3600
    (OUT / "staged_bc.json").write_text(json.dumps(nr._finite(record), indent=1) + "\n")


if __name__ == "__main__":
    main()
