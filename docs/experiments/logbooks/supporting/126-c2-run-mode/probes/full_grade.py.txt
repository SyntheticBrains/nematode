"""The guarded wide probe's answer graded in full on v4 (description, discarded seeds).

Generation 250 of `guarded_wide.json`, on `staged_bc.py`'s model: 5 populations of 200 at
720,000,000 + 200·g + i on all four of v4's assays with presence (every family, local search and
chemotaxis included), and the three mutants on the small lawn on each population's animals, each
graded on its difference from that population's wild type on its v4 band. The mutant encodings are
C.2's registered ones: tph-1 and pdfr-1 by their knockout genes, tax-4 by zeroing the seven tax-4
classes' sensory gains. Run from the repository root.
"""
import json
import sys
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np
import torch

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c2-run-mode")
import benchmark_grade as bg
import c2_encoder_probe as p
import npe_recovery as nr
import staged_bc as b
from quantumnematode.connectome import encoder as enc

BASE = 720_000_000
Z = json.loads(Path("build/c2-run-mode/guarded_wide.json").read_text())["answer"]


def controller(z, mutant=None):
    from quantumnematode.evaluation.controllers import BatchedConnectome

    brain = p.encoded_brain(p.KNOCKOUT_GENES.get(mutant or "", ()))
    units = b.to_encoder(z)
    if mutant == "tax-4":
        units = list(enc.tax4_vector(brain.topology.modulator_spec, units, b.TYING))
    enc.write_encoder(brain, units)
    brain.topology.probe_centres = torch.from_numpy(-1.0 + 2.0 * np.asarray(z[b.N_ENCODER:], dtype=np.float32))
    return lambda _path: BatchedConnectome(brain)


def wild(g):
    bands = bg.load_bands_v4()
    seeds = p.population_seeds(BASE, g)
    populations, chemotaxis = bg.simulate_v2(p.ARM, controller(Z), seeds, bands)
    graded = bg.grade_populations_v4(populations, chemotaxis, bands, presence=True)
    clf = bg.classifier(bands)
    values = bg.family_values("small_lawn", populations["small_lawn"], clf)
    return g, bg._finite({
        "families": {f: {n: {"value": r["value"], "z": r["z"], "passes": r["passes"]} for n, r in rs.items()} for f, rs in graded["families"].items()},
        "family_passes": graded["family_passes"], "wild_type_passes": graded["wild_type_passes"],
        "local_search_presence": graded.get("local_search_presence"), "chemotaxis_presence": graded.get("chemotaxis_presence"),
        "D": graded["D"], "D_fit": graded["D_fit"], "reversal_share": graded["reversal_share"],
        "small_lawn_values": values,
    })


def mutant(item):
    g, name, wild_values = item
    bands = bg.load_bands_v4()
    clf = bg.classifier(bands)
    mutated, _ = bg.simulate_v2(p.ARM, controller(Z, name), p.population_seeds(BASE, g), bands, assays=("small_lawn",))
    values = bg.family_values("small_lawn", mutated["small_lawn"], clf)
    return g, name, p.mutant_reading(name, wild_values, values, bands, p.ANIMALS)


def main():
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        wilds = dict(pool.map(wild, range(5)))
        jobs = [(g, m, wilds[g]["small_lawn_values"]) for g in range(5) for m in p.MUTANTS]
        mutants = list(pool.map(mutant, jobs))
    for g, name, reading in mutants:
        wilds[g].setdefault("mutants", {})[name] = reading
    passing = sum(bool(w["wild_type_passes"]) for w in wilds.values())
    record = {"description": __doc__, "answer": Z, "populations": wilds, "populations_passing": passing, "passes_wild_type": passing >= 3}
    Path("build/c2-run-mode/full_grade.json").write_text(json.dumps(bg._finite(record), indent=1) + "\n")
    for g, w in sorted(wilds.items()):
        fails = [f"{f}/{n}" for f, rs in w["families"].items() for n, r in rs.items() if not r["passes"]]
        print(g, "wild type passes", w["wild_type_passes"], "D", round(w["D"], 2), "D_fit", round(w["D_fit"], 2), "fails", fails,
              "LS present", (w["local_search_presence"] or {}).get("present"), "chem present", (w["chemotaxis_presence"] or {}).get("present"),
              "mutants", {m: r["passes"] for m, r in w["mutants"].items()}, flush=True)
    print("populations passing", passing, "of 5")


if __name__ == "__main__":
    main()
