"""C.1's 26-parameter model on benchmark v4 with its states ordered, from the box centre.

Description, discarded seeds. The unordered box-centre probe (640 M) fell into a basin whose state 0
(dwelling) is the fast one, where r_dwell_on_food acts on the wrong state. Here the dwelling speed
is the roaming speed times a fraction, log-uniform on [0.025, 1] (the unit coordinate that held
v_dwell), so state 0 is always the slower; every other parameter's box is unchanged. From the box
centre at sigma0 0.2; CMA 9448; fit 644,000,000; graded on 646,000,000. v4's ranking score, C.1b's
doubly guarded stop tightened (every median in the last 40 generations below 100), cap 300; graded
on v4 with presence on 5 populations. Reaches: at least 3 of 5 pass. Run from the repository root.
"""

import json
import math
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c1-v4")
import benchmark_grade as bg
import c1_v3_fit as c1v3
import c1b_fit as c1b
import cma_recovery as cr
import mechanisms_v4 as m
import npe_recovery as nr
import readiness_v4 as r

OUT = Path("build/c1-v4")
CAP = 300
LOW, HIGH = 1e-3, 0.5
RUN = {"cma": 9448, "fit": 644_000_000, "grade": 646_000_000, "sigma0": 0.2}
RATIO_LOW = 0.025
TIGHT = 100.0


def _log_unit(value):
    return math.log(value / LOW) / math.log(HIGH / LOW)


def parameters(z):
    z = np.asarray(z, dtype=float)
    out = c1v3.parameters(z[: c1v3.DIMENSION])
    names = list(out)
    ratio = math.exp(math.log(RATIO_LOW) * (1.0 - z[names.index("v_dwell")]))
    out["v_dwell"] = out["v_roam"] * ratio
    out["r_dwell_on_food"] = float(math.exp(math.log(LOW) + z[c1v3.DIMENSION] * math.log(HIGH / LOW)))
    return out


def simulate(k, z, seed_base):
    seeds = [seed_base + nr.ANIMALS * k + i for i in range(nr.ANIMALS)]
    start = time.perf_counter()
    populations, chemotaxis, bands = m._populations(parameters(z), seeds)
    graded = bg.grade_populations_v4(populations, chemotaxis, bands, presence=False)
    share = graded["reversal_share"]
    return nr._finite({
        "k": k, "z": list(map(float, z)),
        "statistics": {f"{f}/{n}": v["value"] for f, rs in graded["families"].items() for n, v in rs.items()},
        "D": graded["D"], "D_fit": graded["D_fit"],
        "reversal_share": None if share is None else share["value"],
        "seconds": time.perf_counter() - start,
    })


def grade(item):
    z, base, g = item
    populations, chemotaxis, bands = m._populations(parameters(z), [base + nr.ANIMALS * g + i for i in range(nr.ANIMALS)])
    graded = bg.grade_populations_v4(populations, chemotaxis, bands, presence=True)
    return bg._finite({
        "g": g, "wild_type_passes": graded["wild_type_passes"], "family_passes": graded["family_passes"],
        "D": graded["D"], "D_fit": graded["D_fit"],
        "failing": [f"{f}/{n}" for f, rs in graded["families"].items() for n, v in rs.items() if not v["passes"]],
        "z": {f"{f}/{n}": v["z"] for f, rs in graded["families"].items() for n, v in rs.items()},
        "values": {f"{f}/{n}": v["value"] for f, rs in graded["families"].items() for n, v in rs.items()},
        "chemotaxis_present": graded["chemotaxis_presence"]["present"],
        "local_search_present": graded["local_search_presence"]["present"],
        "reversal_share": graded["reversal_share"]["value"] if graded["reversal_share"] else None,
    })


def x0():
    return [0.5] * (c1v3.DIMENSION + 1)


def tight_stop(medians, sigma):
    """C.1b's doubly guarded plateau, and every median in the last 40 generations below 100."""
    return c1b.defined_stop(medians, sigma) and max(medians[-2 * cr.PLATEAU_WINDOW :]) < TIGHT


def start(seed):
    import cma

    return cma.CMAEvolutionStrategy(x0(), RUN["sigma0"], {"CMA_diagonal": True, "bounds": [0.0, 1.0], "popsize": cr.POPSIZE, "seed": seed, "verbose": -9})


def main():
    out = OUT / "ordered_box.json"
    t0 = time.time()
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        history = cr.run_fit(
            {}, fit=0, cma_seed=RUN["cma"], seed_base=RUN["fit"], cap=CAP, use_rule=True,
            state_dir=OUT / "ordered_box", pool=pool, scales={}, objective=r.V4BandObjective(),
            simulate=simulate, stop=tight_stop, strategy_factory=start,
        )
        answer = history["answers"][-1]
        grades = list(pool.map(grade, [(answer, RUN["grade"], g) for g in range(5)]))
    passing = sum(bool(g["wild_type_passes"]) for g in grades)
    record = {
        "description": "Description: C.1 on benchmark v4 26 parameters, states ordered, from the box centre; discarded seeds.",
        **RUN, "cap": CAP, "stop": history.get("stop"), "generations": len(history["medians"]),
        "medians": history["medians"], "sigmas": history["sigmas"], "answer": answer,
        "parameters": parameters(answer), "populations_passing": passing, "reaches": passing >= 3,
        "grades": grades, "wall_hours": (time.time() - t0) / 3600,
    }
    out.write_text(json.dumps(bg._finite(record), indent=2) + "\n")
    print(json.dumps({k: record[k] for k in ("stop", "generations", "populations_passing", "reaches")}),
          "D", [round(g["D"], 2) for g in grades], flush=True)
    for g in grades:
        print("  ", g["g"], g["wild_type_passes"], g["failing"], flush=True)


if __name__ == "__main__":
    main()
