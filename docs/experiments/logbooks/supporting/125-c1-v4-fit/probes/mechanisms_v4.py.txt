"""Two candidate mechanisms for C.1 on benchmark v4: probes (description, discarded seeds).

The readiness probe found C.1's 25-parameter model short on on-lawn onsets: it dwells at a
near-constant 0.013 mm/s, so reversals while dwelling follow no forward step the detector sees.

coupling: 26 parameters, reversal_speed_coupling on [0, 3] (v2's reachability worm's box);
          x0 = the readiness warm answer with the coupling at 0; CMA 9443; fit 628,000,000;
          graded on 630,000,000.
pauses:   27 parameters, dwell_pause_share on [0, 0.9] and dwell_pause_s on [1, 20];
          x0 = the readiness warm answer with v_dwell doubled (its mean dwelling speed kept) and
          half the dwelling seconds paused in 3-s pauses; CMA 9444; fit 629,000,000; graded on
          631,000,000.
Both: sigma0 0.1, v4's ranking score against its band centres at n = 200, C.1b's doubly guarded
stop, cap 300; the answer graded on v4 with presence on 5 populations. Reaches: at least 3 of 5
pass. Run from the repository root.
"""

import json
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c1-v4")
import benchmark_grade as bg
import c1_v3_fit as c1v3
import c1b_fit as c1b
import cma_recovery as cr
import npe_recovery as nr
import readiness_v4 as r

OUT = Path("build/c1-v4")
CAP = 300
EXTRA = {
    "coupling": {"reversal_speed_coupling": (0.0, 3.0)},
    "pauses": {"dwell_pause_share": (0.0, 0.9), "dwell_pause_s": (1.0, 20.0)},
}
RUNS = {
    "coupling": {"cma": 9443, "fit": 628_000_000, "grade": 630_000_000, "sigma0": 0.1},
    "pauses": {"cma": 9444, "fit": 629_000_000, "grade": 631_000_000, "sigma0": 0.1},
}


def parameters(z, name):
    z = np.asarray(z, dtype=float)
    out = c1v3.parameters(z[: c1v3.DIMENSION])
    for value, (key, (low, high)) in zip(z[c1v3.DIMENSION :], EXTRA[name].items(), strict=True):
        out[key] = float(low + value * (high - low))
    return out


def _populations(params, seeds, assays=None):
    from quantumnematode.evaluation.controllers import BatchedMechanistic

    bands = bg.load_bands_v4()
    populations, chemotaxis = bg.simulate_v2("mlp", lambda _p: BatchedMechanistic(params), seeds, bands, assays)
    return populations, chemotaxis, bands


def _simulate(k, z, seed_base, name):
    seeds = [seed_base + nr.ANIMALS * k + i for i in range(nr.ANIMALS)]
    start = time.perf_counter()
    populations, chemotaxis, bands = _populations(parameters(z, name), seeds)
    graded = bg.grade_populations_v4(populations, chemotaxis, bands, presence=False)
    share = graded["reversal_share"]
    return nr._finite({
        "k": k, "z": list(map(float, z)),
        "statistics": {f"{f}/{n}": v["value"] for f, rs in graded["families"].items() for n, v in rs.items()},
        "D": graded["D"], "D_fit": graded["D_fit"],
        "reversal_share": None if share is None else share["value"],
        "seconds": time.perf_counter() - start,
    })


def simulate_coupling(k, z, seed_base):
    return _simulate(k, z, seed_base, "coupling")


def simulate_pauses(k, z, seed_base):
    return _simulate(k, z, seed_base, "pauses")


def grade(item):
    z, base, g, name = item
    populations, chemotaxis, bands = _populations(parameters(z, name), [base + nr.ANIMALS * g + i for i in range(nr.ANIMALS)])
    graded = bg.grade_populations_v4(populations, chemotaxis, bands, presence=True)
    return bg._finite({
        "g": g, "wild_type_passes": graded["wild_type_passes"], "family_passes": graded["family_passes"],
        "D": graded["D"], "D_fit": graded["D_fit"],
        "failing": [f"{f}/{n}" for f, rs in graded["families"].items() for n, v in rs.items() if not v["passes"]],
        "z": {f"{f}/{n}": v["z"] for f, rs in graded["families"].items() for n, v in rs.items()},
        "values": {f"{f}/{n}": v["value"] for f, rs in graded["families"].items() for n, v in rs.items()},
        "chemotaxis_present": graded["chemotaxis_presence"]["present"],
        "local_search_present": graded["local_search_presence"]["present"],
        "reversal_share": graded["reversal_share"]["value"] if graded["reversal_share"] else None,
    })


def _warm():
    return list(json.loads((OUT / "readiness_warm.json").read_text())["answer"])


def x0(name):
    z = _warm()
    if name == "coupling":
        return [*z, 0.0]
    names = list(c1v3.parameters(z))
    i = names.index("v_dwell")
    kind, low, high = nr.PRIOR["v_dwell"]
    v = low + z[i] * (high - low)
    z[i] = min(1.0, (2.0 * v - low) / (high - low))
    return [*z, 0.5 / 0.9, (3.0 - 1.0) / 19.0]


def _start(name, seed):
    import cma

    return cma.CMAEvolutionStrategy(x0(name), RUNS[name]["sigma0"], {"CMA_diagonal": True, "bounds": [0.0, 1.0], "popsize": cr.POPSIZE, "seed": seed, "verbose": -9})


def start_coupling(seed):
    return _start("coupling", seed)


def start_pauses(seed):
    return _start("pauses", seed)


def run(name, pool):
    p = RUNS[name]
    out = OUT / f"mechanism_{name}.json"
    if out.exists():
        return
    t0 = time.time()
    history = cr.run_fit(
        {}, fit=0, cma_seed=p["cma"], seed_base=p["fit"], cap=CAP, use_rule=True,
        state_dir=OUT / f"mechanism_{name}", pool=pool, scales={}, objective=r.V4BandObjective(),
        simulate=simulate_coupling if name == "coupling" else simulate_pauses, stop=c1b.defined_stop,
        strategy_factory=start_coupling if name == "coupling" else start_pauses,
    )
    answer = history["answers"][-1]
    grades = list(pool.map(grade, [(answer, p["grade"], g, name) for g in range(5)]))
    passing = sum(bool(g["wild_type_passes"]) for g in grades)
    record = {
        "description": f"Description: C.1 on benchmark v4 with the '{name}' mechanism; discarded seeds.",
        **p, "extra": {k: list(v) for k, v in EXTRA[name].items()}, "cap": CAP, "stop": history.get("stop"),
        "generations": len(history["medians"]), "medians": history["medians"], "sigmas": history["sigmas"],
        "answer": answer, "parameters": parameters(answer, name), "populations_passing": passing,
        "reaches": passing >= 3, "grades": grades, "wall_hours": (time.time() - t0) / 3600,
    }
    out.write_text(json.dumps(bg._finite(record), indent=2) + "\n")
    print(name, json.dumps({k: record[k] for k in ("stop", "generations", "populations_passing", "reaches")}),
          "D", [round(g["D"], 2) for g in grades], flush=True)
    for g in grades:
        print("  ", g["g"], g["wild_type_passes"], g["failing"], flush=True)


def main():
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        for name in sys.argv[1:] or list(RUNS):
            run(name, pool)


if __name__ == "__main__":
    main()
