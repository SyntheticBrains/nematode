def earlier(path: str) -> dict[str, tuple[str, str]]:
    text = Path(path).read_text()
    namespace: dict = {}
    exec(text.split("\nonly = sys.argv[1:]")[0], namespace)  # noqa: S102 - our own harness
    return namespace["MUTANTS"]


MUTANTS = {
    **{f"[presence] {k}" if not k.startswith("[") else k: v for k, v in NEW.items()},
    **{k: v for k, v in earlier("build/c2-fit/mutants_fresh.py").items()},
}

only = sys.argv[1:]
original = SOURCE.read_text()
counts = {"caught": 0, "MISSED": 0, "n/a": 0}
try:
    for name, (old, new) in MUTANTS.items():
        if only and name not in only:
            continue
        n = original.count(old)
        if n != 1:
            print(f"{name:75s} n/a (pattern found {n} times)", flush=True)
            counts["n/a"] += 1
            continue
        SOURCE.write_text(original.replace(old, new))
        result = subprocess.run(
            [sys.executable, "-m", "pytest", TEST, "-q", "-m", "not slow", "-n", "8", "--no-cov",
             "-p", "no:cacheprovider", "-p", "no:randomly", "-x"],
            capture_output=True, text=True, check=False,
        )
        caught = result.returncode != 0
        failed = [l.split("::")[-1].split(" ")[0] for l in result.stdout.splitlines() if l.startswith("FAILED")]
        counts["caught" if caught else "MISSED"] += 1
        print(f"{name:75s} {'caught' if caught else 'MISSED'} {failed[:2]}", flush=True)
finally:
    SOURCE.write_text(original)
assert SOURCE.read_text() == original
print(counts, flush=True)
