"""Can a start that does not hold the lawn be made to? 2a from the pilot's start with a lawn guard (description; discarded)

Every 2a lineage that reached the basin held the small lawn early (time on the lawn at least 0.975
over generations 40-60) and the two that did not never held it (the pilot 0.80-0.92 over all 300
generations, stalled at D_fit 3.1-3.3; screen a 0.89-0.96). That is a correlation across six
lineages. This asks whether it is causal: the pilot's 2a again, from the same stage-1 answer with the
speed noise's timescale at 1 s, sigma0 0.1, population 48, 200 animals, the registered arm and
objective (c2_fit's), **plus one guard**: 300 x (0.95 - t) / 0.95 while the candidate's time on the
small lawn t is below 0.95, graded like the responsiveness guard, so holding the lawn comes before
everything else and costs nothing once held.

**The rules, fixed before it runs** (2026-11-12):
- it **holds the lawn** when the mean time on the lawn at generations 40, 50 and 60 is at least 0.95;
- it **arrives** when the mean of three consecutive readouts has time on the lawn at least 0.97 and
  D_fit at most 2.5, the registered bar at 2a's end, and stops there; otherwise read at generation 300;
- holds and arrives: lawn-holding is causal and a guard is a remedy; holds and does not arrive:
  holding the lawn is not enough from this start; does not hold: the guard cannot rescue it.

Seeds (discarded): candidates 1,050,000,000 + 2,000,000 + 200 x (48 x (g - 1) + m) + i, readouts
1,055,000,000 + i (its own, with a generation-0 readout of the start). CMA 9525. 8 workers.
"""

import json
import pickle
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c2-fit")
import benchmark_grade as bg
import c2_fit as fit
import npe_recovery as nr
from continue_screen import BAR, _job, reading
from quantumnematode.connectome import encoder as enc

BASE, CMA, POPSIZE, WORKERS, LAST = 1_050_000_000, 9525, 48, 8, 300
STRIDE, READOUT = 2_000_000, 5_000_000
HOLDS, GUARD, GUARD_BELOW = 0.95, 300.0, 0.95
LAWN = "small_lawn/time_on_lawn"
OUT = Path("build/c2-fit/guard_probe")


def lawn_guard(record):
    t = record["statistics"].get(LAWN)
    t = 0.0 if t is None else float(t)
    return GUARD * (GUARD_BELOW - t) / GUARD_BELOW if t < GUARD_BELOW else 0.0


def main():
    OUT.mkdir(parents=True, exist_ok=True)
    state_path = OUT / "state.pkl"
    bands = bg.load_bands_v4()
    truth = fit.band_truth(bands)
    names = fit.entries()
    objective = fit.Objective(truth, bands, families=fit.FIT_FAMILIES, mutants=())
    if state_path.exists():
        state = pickle.loads(state_path.read_bytes())
    else:
        pilot = json.loads(Path("docs/experiments/logbooks/supporting/133-c2-fit/pilot.json").read_text())
        x0 = [float(v) for v in pilot["answers"]["1"]]
        assert len(x0) == len(names) == 137
        ts = names.index(next(e for e in names if e.name == enc.SPEED_NOISE_ENTRY))
        x0[ts] = names[ts].unit(fit.TIMESCALE_AT_2A_S)
        state = {"es": fit.fresh(x0, 0.1, CMA, POPSIZE), "medians": [], "guarded": [], "readouts": {},
                 "holds": None, "verdict": None}
    es = state["es"]
    readout_seeds = list(range(BASE + READOUT, BASE + READOUT + 200))
    t0 = time.time()
    with ProcessPoolExecutor(max_workers=WORKERS, initializer=nr._worker_setup) as pool:
        if 0 not in state["readouts"]:
            r = state["readouts"][0] = fit.readout(pool, list(np.clip(es.result.xfavorite, 0, 1)), readout_seeds, truth)
            print(f"guard probe readout 0: D_fit {r['D_fit']:.2f} on-lawn {r['time_on_lawn']:.3f} rev {r['spread']['reversal']:.2f}", flush=True)
        while len(state["medians"]) < LAST and state["verdict"] is None:
            g = len(state["medians"]) + 1
            points = es.ask()
            jobs = []
            for m, p in enumerate(points):
                start = BASE + STRIDE + 200 * (POPSIZE * (g - 1) + m)
                jobs.append((list(range(start, start + 200)), list(map(float, np.clip(p, 0, 1)))))
            records = list(pool.map(_job, jobs))
            guards = [lawn_guard(r) for r in records]
            scores = [objective(r) + q for r, q in zip(records, guards)]
            es.tell(points, scores)
            state["medians"].append(float(np.median(scores)))
            state["guarded"].append(float(np.mean([q > 0 for q in guards])))
            if g % 10 == 0:
                r = state["readouts"][g] = fit.readout(pool, list(np.clip(es.result.xfavorite, 0, 1)), readout_seeds, truth)
                print(f"guard probe readout {g}: D_fit {r['D_fit']:.2f} on-lawn {r['time_on_lawn']:.3f} rev {r['spread']['reversal']:.2f}", flush=True)
                if g == 60:
                    lawn = float(np.mean([state["readouts"][k]["time_on_lawn"] for k in (40, 50, 60)]))
                    state["holds"] = {"time_on_lawn": lawn, "holds": lawn >= HOLDS}
                    print(f"guard probe lawn-holding at 40-60: {state['holds']}", flush=True)
                if g >= 30:
                    three = reading(state["readouts"], (g - 20, g - 10, g))
                    if three["arrives"]:
                        state["verdict"] = {**three, "verdict": "arrives"}
            print(f"guard probe gen {g}: median {state['medians'][-1]:.2f} guarded {state['guarded'][-1]:.2f} sigma {es.sigma:.4f} {time.time() - t0:.0f} s", flush=True)
            state_path.write_bytes(pickle.dumps(state))
    if state["verdict"] is None:
        g = len(state["medians"])
        state["verdict"] = {**reading(state["readouts"], (g - 20, g - 10, g)), "verdict": "does not arrive"}
    record = {"description": __doc__, "base": BASE, "cma": CMA, "bar": BAR, "holds_bar": HOLDS,
              "guard": {"weight": GUARD, "below": GUARD_BELOW}, "holds": state["holds"], "reading": state["verdict"],
              "medians": state["medians"], "guarded_share": state["guarded"],
              "readouts": {str(k): v for k, v in state["readouts"].items()},
              "answer": list(map(float, np.clip(es.result.xfavorite, 0, 1)))}
    (OUT.parent / "guard_probe.json").write_text(json.dumps(nr._finite(record), indent=1) + "\n")
    print(f"guard probe done: holds {state['holds']} reading {state['verdict']}", flush=True)


if __name__ == "__main__":
    main()
