"""Do the slow screens arrive late, or stay stuck? (description; discarded seeds)

Screens (a) and (b) missed the pace rule at 2a's generation 100 (D_fit 4.82 and 4.37), but both were
still improving by about 0.5 every 20 generations, where the pilot's 2a sat flat at 3.1-3.3 to its
generation 300. So two of the three from-scratch first halves are undetermined, and the share that
reach the basin within the registered 2a (300 generations) is somewhere between 1 in 4 and 3 in 4.

The screens kept no search state, so this is **not an exact continuation**: sep-CMA starts again from
the screen's generation-100 answer at the screen's last step size (a 0.0754, b 0.0852), isotropic, on
the registered arm and objective (c2_fit's: the ranking score on D_fit's families with both guards),
population 48, 200 animals, 8 workers, for up to 150 generations (250 in all).

**The rule, fixed before either runs** (2026-11-11): it **arrives** when the mean of three consecutive
readouts has time on the lawn at least 0.97 and D_fit at most 2.5, the registered bar at 2a's end, and
stops there. Otherwise it is read at generation 150 on the mean of its last three readouts: **stuck**.

Seeds (discarded, within each screen's own range): candidates continue the screen's 2a numbering,
base + 2,000,000 + 200 x (48 x (100 + g - 1) + m) + i for g = 1..150, below the readouts at
base + 5,000,000 + i, which are the screen's own so the readouts compare. CMA 9521 (a), 9522 (b).
Run from the repository root: ``continue_screen.py a`` and ``continue_screen.py b``.
"""

import json
import pickle
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
import benchmark_grade as bg
import c2_fit as fit
import npe_recovery as nr

SCREENS = {
    "a": {"base": 1_010_000_000, "cma": 9521, "sigma0": 0.0754},
    "b": {"base": 1_020_000_000, "cma": 9522, "sigma0": 0.0852},
}
STRIDE, READOUT, DONE, GENERATIONS, POPSIZE, WORKERS = 2_000_000, 5_000_000, 100, 150, 48, 8
BAR = {"time_on_lawn": 0.97, "D_fit": 2.5}
OUT = Path("build/c2-fit/continue")


def _job(item):
    seeds, z = item
    return fit.simulate_point(z, seeds, assays=fit.FIT_ASSAYS, mutants=())


def reading(readouts, gens):
    last = [readouts[g] for g in gens]
    out = {
        "generations": list(gens),
        "time_on_lawn": float(np.mean([r["time_on_lawn"] for r in last])),
        "D_fit": float(np.mean([r["D_fit"] for r in last])),
    }
    out["arrives"] = out["time_on_lawn"] >= BAR["time_on_lawn"] and out["D_fit"] <= BAR["D_fit"]
    return out


def main(name):
    cfg = SCREENS[name]
    out = OUT / name
    out.mkdir(parents=True, exist_ok=True)
    state_path = out / "state.pkl"
    bands = bg.load_bands_v4()
    truth = fit.band_truth(bands)
    objective = fit.Objective(truth, bands, families=fit.FIT_FAMILIES, mutants=())
    screen = json.loads(Path(f"build/c2-fit/screens/{name}.json").read_text())
    x0 = [float(v) for v in screen["answers"]["1"]]
    assert len(x0) == len(fit.entries()) == 137
    if state_path.exists():
        state = pickle.loads(state_path.read_bytes())
    else:
        state = {"es": fit.fresh(x0, cfg["sigma0"], cfg["cma"], POPSIZE), "medians": [], "readouts": {}}
    es = state["es"]
    readout_seeds = list(range(cfg["base"] + READOUT, cfg["base"] + READOUT + 200))
    t0 = time.time()
    verdict = None
    with ProcessPoolExecutor(max_workers=WORKERS, initializer=nr._worker_setup) as pool:
        while len(state["medians"]) < GENERATIONS and verdict is None:
            g = len(state["medians"]) + 1
            points = es.ask()
            jobs = []
            for m, p in enumerate(points):
                start = cfg["base"] + STRIDE + 200 * (POPSIZE * (DONE + g - 1) + m)
                jobs.append((list(range(start, start + 200)), list(map(float, np.clip(p, 0, 1)))))
            records = list(pool.map(_job, jobs))
            scores = [objective(r) for r in records]
            es.tell(points, scores)
            state["medians"].append(float(np.median(scores)))
            if g % 10 == 0:
                r = fit.readout(pool, list(np.clip(es.result.xfavorite, 0, 1)), readout_seeds, truth)
                state["readouts"][g] = r
                print(f"continue {name} readout {DONE + g}: D_fit {r['D_fit']:.2f} on-lawn {r['time_on_lawn']:.3f} "
                      f"rev {r['spread']['reversal']:.2f}", flush=True)
                if g >= 30:
                    three = reading(state["readouts"], (g - 20, g - 10, g))
                    if three["arrives"]:
                        verdict = {**three, "verdict": "arrives"}
            print(f"continue {name} gen {DONE + g}: median {state['medians'][-1]:.2f} sigma {es.sigma:.4f} {time.time() - t0:.0f} s", flush=True)
            state_path.write_bytes(pickle.dumps(state))
    if verdict is None:
        g = len(state["medians"])
        verdict = {**reading(state["readouts"], (g - 20, g - 10, g)), "verdict": "stuck"}
    record = {"description": __doc__, "screen": name, **cfg, "bar": BAR, "reading": verdict,
              "medians": state["medians"], "readouts": {str(DONE + k): v for k, v in state["readouts"].items()},
              "answer": list(map(float, np.clip(es.result.xfavorite, 0, 1)))}
    (OUT / f"{name}.json").write_text(json.dumps(nr._finite(record), indent=1) + "\n")
    print(f"continue {name} done: {verdict}", flush=True)


if __name__ == "__main__":
    main(sys.argv[1])
