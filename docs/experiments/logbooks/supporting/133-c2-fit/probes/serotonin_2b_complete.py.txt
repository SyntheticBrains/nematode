"""Read-only: every recovery 2b candidate's tph-1 roaming difference, from the completed log.

The diagnosis of 2026-11-23 ran while 2b was at about generation 160. This reads the whole of 2b's
candidate log, after the recovery ended: how many candidates, their generations, and the largest
tph-1 roaming difference among them. No simulation.

Usage: serotonin_2b_complete.py.txt <out.json>   (run from the repository root)
"""

import hashlib
import json
import sys
from pathlib import Path

LOG = Path("build/c2-fit/recovery/stage_2b.jsonl")  # off-repo, beside the recovery's state

values, generations = [], set()
with LOG.open() as handle:
    for line in handle:
        record = json.loads(line)
        generations.add(int(record["generation"]))
        t = ((record.get("delta") or {}).get("tph-1") or {}).get("roaming_fraction")
        if t is not None:
            values.append(float(t))
out = {
    "description": __doc__,
    "log_sha256": hashlib.sha256(LOG.read_bytes()).hexdigest(),
    "candidates": len(values),
    "generations": [min(generations), max(generations)],
    "max_tph1_roaming_difference": max(values),
    "at_least_0.05": sum(v >= 0.05 for v in values),
}
Path(sys.argv[1]).write_text(json.dumps(out, indent=1) + "\n")
print(json.dumps({k: v for k, v in out.items() if k != "description"}, indent=1))
