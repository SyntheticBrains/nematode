"""Do real worms' on-lawn reversals differ between dwelling and roaming? (description; band cohorts)

The primitives give every reversal that starts on food one set of kinematics (Scheer's held-out
cohorts pooled: 2.05 s at 0.069 mm/s). Here the band cohorts' wild type (PD1074 od2, small lawn) is
read by state: each reversal bout (the kinematics check's rule) that starts on the lawn (midbody
inside the boundary) takes the roaming/dwelling label of its 10-s bin; reported per state: bouts
per on-lawn minute in that state, mean duration, and mean speed (path over the bout's whole
seconds, as the kinematics check reads it).
"""
import json, sys
import numpy as np
sys.path.insert(0, "scripts/analysis")
import primitive_kinematics_check as pk
from quantumnematode.benchmark import grading as g
from quantumnematode.benchmark.references import load_scheer_tracks

t = load_scheer_tracks()
gen = np.array([t.genotypes[k] for k in t.genotype])
rows = np.flatnonzero((gen == "PD1074_od2_Data") & (t.side == "bands"))
revf = t.flag("reversal")[rows]
acc = {0: {"bouts": 0, "seconds": 0, "dur": [], "speed": []}, 1: {"bouts": 0, "seconds": 0, "dur": [], "speed": []}}
for j, i in enumerate(rows):
    x, y = t.x[i].astype(float), t.y[i].astype(float)
    on = g.real_on_lawn(x, y, t.boundary_radius[i])
    lab = np.repeat(t.labels[i], 10)[: len(x)]
    for state in (0, 1):
        acc[state]["seconds"] += int(((lab == state) & on).sum())
    step = np.hypot(np.diff(x), np.diff(y))
    for a, b in pk.bouts(revf[j], filtered=True):
        s0 = a // 3
        if s0 >= len(x) or not on[s0] or lab[s0] not in (0, 1):
            continue
        st = int(lab[s0]); acc[st]["bouts"] += 1; acc[st]["dur"].append((b - a) / 3.0)
        first, last = -(-a // 3), b // 3
        if last > first and last <= len(step):
            path = step[first:last]
            if np.isfinite(path).all():
                acc[st]["speed"].append(float(path.sum() / (last - first)))
out = {"description": __doc__}
for st, name in ((0, "dwelling"), (1, "roaming")):
    a = acc[st]
    out[name] = {"bouts_per_min": 60.0 * a["bouts"] / a["seconds"], "duration_s": float(np.mean(a["dur"])),
                 "speed_mm_s": float(np.mean(a["speed"])), "bouts": a["bouts"], "speed_samples": len(a["speed"])}
    print(name, {k: round(v, 4) if isinstance(v, float) else v for k, v in out[name].items()})
print("primitives on food: duration 2.047 s, speed 0.069 mm/s")
json.dump(out, open("build/c2-change-b/reversal_by_state.json", "w"), indent=1)
