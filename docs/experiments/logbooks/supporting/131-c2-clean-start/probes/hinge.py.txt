"""A continuation scored the way the verdict is (description; discarded seeds).

Pre-approved for one case: the curriculum's stage-3 verdict fails, the mutants pass, and the wild
type fails on the on-lawn onset rate alone. The fit minimises a distance, an RMS over 24
statistics, while the verdict is pass or fail on each, so a statistic a few tenths past its bar
barely moves the score (`onset_gap.json`: the fit's true on-lawn reversal starts are 3.0 a minute
against real worms' 4.3, and its detected rate 1.5 against 2.3).

From the stage-3 answer, sep-CMA on stage 3's objective (the six families plus tph-1's, pdfr-1's
and tax-4's lawn differences, the mutants together half of it, the graded responsiveness guard and
the saturation guard) plus a hinge: 5 x the sum over statistics of max(0, |z| - bar)^2, the bar
0.3 inside each pass rule (2.78 for a wild-type statistic, against 3.0781; 1.94 for a mutant's,
against 2.24), z on the bands' scales at 200 animals as the grade reads them, compressed as the
ranking score compresses it (10·asinh(z/10)) so a statistic far off cannot swamp the rest. The
search starts with stage 3's own per-parameter step sizes (its state saved at generation 132,
`curriculum/state_stage3_gen132.pkl`: sigma 0.0224, steps 0.008-0.042), since the curriculum drops a
stage's search state when the stage ends. A first launch with a uniform sigma0 of 0.03 was stopped
after 2 generations (`hinge_isotropic_aborted/`): it moved the sensitive parameters 3.5x further
than stage 3 had, pushing most candidates' statistics far past their bars (medians 417 and 709;
`hinge_candidates.py`). 50
generations, population 48 at 200 animals on 16 workers, readouts every 10 generations. The
verdict, as stage 3's: the last answer graded with presence on 5 fresh populations passes the wild
type in at least 3 and tph-1 and tax-4 each in at least 3.

Seeds: candidates 860,000,000 + 200·k + i, CMA 9500, readouts 861,000,000 + i, the grade
862,000,000 + 200·g + i. Run from the repository root: `hinge.py [smoke]`.
"""

import json
import math
import pickle
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c2-change-b")
import c2_encoder_probe as p
import curriculum as C
import npe_recovery as nr
import probes as P

SMOKE = sys.argv[1:] == ["smoke"]
OUT = Path("build/c2-change-b/hinge_smoke" if SMOKE else "build/c2-change-b/hinge")
BASE, CMA_SEED, READ, GRADE = 860_000_000, 9500, 861_000_000, 862_000_000
POPSIZE, WORKERS, CAP = (4, 2, 1) if SMOKE else (48, 16, 50)
SEARCH_STATE = Path("build/c2-change-b/curriculum/state_stage3_gen132.pkl")
HINGE_WEIGHT, MARGIN, WILD_BAR, MUTANT_BAR = 5.0, 0.3, 3.0781, 2.24
MUTANTS = ("tph-1", "pdfr-1", "tax-4")
C.STAGES.append({"name": "hinge", "base": BASE, "cma": CMA_SEED, "sigma0": None, "cap": CAP, "families": None, "mutants": MUTANTS})
C.GRADE = GRADE
STAGE = len(C.STAGES) - 1


def simulate(k, z):
    return C.simulate(STAGE, k, z)


def _wild(item):
    return C._wild(item)


def _mutant(item):
    return C._mutant(item)


def _compressed(z):
    """The ranking score's compression, 10·asinh(z/10): z to within 1.5% for |z| <= 3."""
    return 10.0 * math.asinh(z / 10.0)


class HingeObjective(P.Objective):
    """Stage 3's objective plus a hinge on every statistic past a bar just inside its pass rule."""

    def __init__(self):
        super().__init__(None, MUTANTS, graded=True)

    def hinge(self, record):
        excess = 0.0
        for key, centre in self.truth["statistics"].items():
            value = record["statistics"].get(key)
            if value is None or not math.isfinite(value):
                continue
            z = _compressed((float(value) - centre) / self.scales[key])
            excess += max(0.0, abs(z) - (WILD_BAR - MARGIN)) ** 2
        for zs in record["mutant_z"].values():
            for z in zs.values():
                if z is not None and math.isfinite(z):
                    excess += max(0.0, abs(_compressed(z)) - (MUTANT_BAR - MARGIN)) ** 2
        return HINGE_WEIGHT * excess

    def __call__(self, record):
        return super().__call__(record) + self.hinge(record)


def start():
    record = json.loads(Path("build/c2-change-b/curriculum.json").read_text())
    if SMOKE:
        return record["z"]
    if record["stage"] != 3 or record["verdict"] is None:
        msg = f"the curriculum has not finished stage 3 (stage {record['stage']}, verdict {record['verdict']})"
        raise SystemExit(msg)
    return record["z"]


def main():
    import cma

    OUT.mkdir(parents=True, exist_ok=True)
    state_path, record_path = OUT / "state.pkl", OUT.parent / ("hinge_smoke.json" if SMOKE else "hinge.json")
    if state_path.exists():
        saved = pickle.loads(state_path.read_bytes())
    else:
        z0 = start()
        stage3 = pickle.loads(SEARCH_STATE.read_bytes())["es"]
        sigma0, stds = float(stage3.sigma), np.asarray(stage3.stds, dtype=float)
        options = {"CMA_diagonal": True, "CMA_stds": (stds / sigma0).tolist(), "bounds": [0.0, 1.0], "popsize": POPSIZE, "seed": CMA_SEED, "verbose": -9}
        saved = {"z0": z0, "sigma0": sigma0, "stds0": stds.tolist(), "es": cma.CMAEvolutionStrategy(z0, sigma0, options),
                 "medians": [], "best": [], "hinge_best": [], "readouts": {}, "verdict": None}
    objective = HingeObjective()
    t0 = time.time()
    with ProcessPoolExecutor(max_workers=WORKERS, initializer=nr._worker_setup) as pool:
        if not saved["readouts"] and not SMOKE:
            saved["readouts"][0] = P.readout(pool, saved["z0"], READ)
            P._print("hinge", 0, saved["readouts"][0])
        es = saved["es"]
        while len(saved["medians"]) < CAP:
            g = len(saved["medians"]) + 1
            points = es.ask()
            records = list(pool.map(simulate, [POPSIZE * (g - 1) + m for m in range(POPSIZE)], [list(map(float, x)) for x in points]))
            scores = [objective(r) for r in records]
            hinges = [objective.hinge(r) for r in records]
            es.tell(points, scores)
            saved["medians"].append(float(np.median(scores))); saved["best"].append(float(np.min(scores)))
            saved["hinge_best"].append(float(hinges[int(np.argmin(scores))]))
            print(f"hinge gen {g}: median {saved['medians'][-1]:.2f} best {saved['best'][-1]:.2f} (hinge {saved['hinge_best'][-1]:.2f}) sigma {es.sigma:.3f} {time.time() - t0:.0f} s", flush=True)
            if g % 10 == 0 and not SMOKE:
                saved["readouts"][g] = P.readout(pool, np.clip(es.result.xfavorite, 0, 1).tolist(), READ)
                P._print("hinge", g, saved["readouts"][g])
            state_path.write_bytes(pickle.dumps(saved))
        z = np.clip(es.result.xfavorite, 0, 1).tolist()
        if not SMOKE:
            wilds = dict(pool.map(_wild, [(g, z) for g in range(5)]))
            for g, m, ok in pool.map(_mutant, [(g, m, z, wilds[g]["small_lawn"]) for g in wilds for m in p.MUTANTS]):
                wilds[g].setdefault("mutants", {})[m] = ok
            wt = sum(bool(w["wild_type_passes"]) for w in wilds.values())
            mut = {m: sum(w["mutants"][m] for w in wilds.values()) for m in p.MUTANTS}
            saved["grade"] = {str(g): {k: v for k, v in w.items() if k != "small_lawn"} for g, w in wilds.items()}
            saved["verdict"] = f"wild type {wt}/5, mutants {mut}: {'passes' if wt >= 3 and mut['tph-1'] >= 3 and mut['tax-4'] >= 3 else 'fails'}"
    saved["z"] = z
    state_path.write_bytes(pickle.dumps(saved))
    record_path.write_text(json.dumps(nr._finite({"description": __doc__, **{k: v for k, v in saved.items() if k != "es"}}), indent=1) + "\n")
    print(f"hinge done: {saved['verdict']}", flush=True)


if __name__ == "__main__":
    main()
