"""Re-grade the curriculum's verdict populations with presence, recording every family's pass (description).

The curriculum's grade kept the wild-type pass and the failing statistics only; presence (local
search and chemotaxis must be present) folds into those families' passes, so it is read here on the
same 5 populations (846,000,000 + 200·g + i) and the same answer.
"""
import json, sys
from concurrent.futures import ProcessPoolExecutor
sys.path.insert(0, "scripts/analysis"); sys.path.insert(0, "build/c2-change-b")
import benchmark_grade as bg, c2_encoder_probe as p, curriculum as C, npe_recovery as nr, probes as P


def job(item):
    g, z = item
    bands = bg.load_bands_v4()
    pops, chem = bg.simulate_v2(p.ARM, P._controller(z), p.population_seeds(C.GRADE, g), bands)
    graded = bg.grade_populations_v4(pops, chem, bands, presence=True)
    return g, nr._finite({
        "wild_type_passes": graded["wild_type_passes"], "family_passes": graded["family_passes"], "D": graded["D"],
        "local_search_present": graded.get("local_search_presence", {}).get("present"),
        "chemotaxis_present": graded.get("chemotaxis_presence", {}).get("present"),
        "statistics": {f"{f}/{n}": {"value": r["value"], "z": r["z"]} for f, rs in graded["families"].items() for n, r in rs.items()},
    })


if __name__ == "__main__":
    z = json.loads(open("build/c2-change-b/curriculum.json").read())["z"]
    with ProcessPoolExecutor(max_workers=5, initializer=nr._worker_setup) as pool:
        out = dict(pool.map(job, [(g, z) for g in range(5)]))
    for g, r in sorted(out.items()):
        print(g, "WT", r["wild_type_passes"], "D", round(r["D"], 2), "presence: search", r["local_search_present"], "chemotaxis", r["chemotaxis_present"],
              "failing families", [f for f, v in r["family_passes"].items() if not v])
    json.dump({"description": __doc__, "populations": out}, open("build/c2-change-b/verify_presence.json", "w"), indent=1)
