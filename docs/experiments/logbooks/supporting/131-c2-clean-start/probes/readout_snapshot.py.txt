"""One full-value readout of the stage-3 search mean, on the curriculum's readout animals (description)."""
import json, pickle, sys
from concurrent.futures import ProcessPoolExecutor
import numpy as np
sys.path.insert(0, "scripts/analysis"); sys.path.insert(0, "build/c2-change-b")
import curriculum as C, npe_recovery as nr, probes as P

if __name__ == "__main__":
    s = pickle.load(open(sys.argv[1], "rb"))
    g = len(s["history"]["3"]["medians"])
    z = np.clip(s["es"].result.xfavorite, 0, 1).tolist()
    with ProcessPoolExecutor(max_workers=4, initializer=nr._worker_setup) as pool:
        r = P.readout(pool, z, C.READ)
    print("generation", g, "D", round(r["D"], 2), "statistics", len(r["statistics"]))
    watch = ["activity/onset_rate_per_min", "kinematics/dwelling_speed_p90", "small_lawn/pre_exit_enrichment",
             "small_lawn/exits_per_on_lawn_hour", "uniform_lawn/roaming_bout_s", "local_search/log_early_late_ratio"]
    for k in watch:
        print(k, {key: round(v, 4) for key, v in r["statistics"][k].items()})
    for m, v in r["mutants"].items():
        print(m, {st: {key: round(x, 3) for key, x in d.items()} for st, d in v["statistics"].items()})
    json.dump({"description": __doc__, "generation": g, "readout": nr._finite(r)}, open("build/c2-change-b/stage3_snapshot.json", "w"), indent=1)
