"""A second continuation, each statistic's target set from its own noise (description; discarded seeds).

The first continuation (`hinge.py`) passed the wild type in 2 of 5 populations with every mutant in
5 of 5, its failures statistics parked just inside their bars. Graded on 15 populations
(`noise_measure.json`) its answer passes the wild type in 8; the spread between populations is
small for most statistics and large for pre-exit enrichment (SD 1.17 in z), whose exits are few.

From that answer, with the first continuation's search state carried over (its sigma and
per-parameter steps at its end), sep-CMA on the same objective: stage 3's (the six families plus
tph-1's, pdfr-1's and tax-4's lawn differences, the mutants half of it, both guards) plus a hinge,
5 x the sum over statistics of max(0, |10·asinh(z/10)| - target)^2, each statistic's target its
pass rule's bar (3.0781 for a wild-type statistic, 2.24 for a mutant's) less 1.65 x its SD between
the 15 populations, so a statistic held at its target passes a population 95% of the time.
50 generations, population 48 at 200 animals on 16 workers, readouts every 10 generations with
every statistic recorded. The verdict as stage 3's: the last answer graded with presence on 5 fresh
populations passes the wild type in at least 3 and tph-1 and tax-4 each in at least 3.

Seeds: candidates 864,000,000 + 200·k + i, CMA 9501, readouts 865,000,000 + i, the grade
866,000,000 + 200·g + i. Run from the repository root.
"""

import json
import math
import pickle
import sys
import time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

sys.path.insert(0, "scripts/analysis")
sys.path.insert(0, "build/c2-change-b")
import c2_encoder_probe as p
import curriculum as C
import npe_recovery as nr
import probes as P

OUT = Path("build/c2-change-b/hinge2")
BASE, CMA_SEED, READ, GRADE = 864_000_000, 9501, 865_000_000, 866_000_000
POPSIZE, WORKERS, CAP = 48, 16, 50
HINGE_WEIGHT, SAFETY, WILD_BAR, MUTANT_BAR = 5.0, 1.65, 3.0781, 2.24
MUTANTS = ("tph-1", "pdfr-1", "tax-4")
START = Path("build/c2-change-b/hinge.json")
SEARCH_STATE = Path("build/c2-change-b/hinge/state.pkl")
NOISE = json.loads(Path("build/c2-change-b/noise_measure.json").read_text())["summary"]
C.STAGES.append({"name": "hinge2", "base": BASE, "cma": CMA_SEED, "sigma0": None, "cap": CAP, "families": None, "mutants": MUTANTS})
C.GRADE = GRADE
STAGE = len(C.STAGES) - 1
TARGETS = {k: (MUTANT_BAR if k.split("/")[0] in MUTANTS else WILD_BAR) - SAFETY * v["sd_z"] for k, v in NOISE.items() if isinstance(v, dict)}


def simulate(k, z):
    return C.simulate(STAGE, k, z)


def _wild(item):
    return C._wild(item)


def _mutant(item):
    return C._mutant(item)


def _compressed(z):
    """The ranking score's compression, 10·asinh(z/10): z to within 1.5% for |z| <= 3."""
    return 10.0 * math.asinh(z / 10.0)


class NoiseHingeObjective(P.Objective):
    """Stage 3's objective plus a hinge on each statistic past its own noise-set target."""

    def __init__(self):
        super().__init__(None, MUTANTS, graded=True)

    def hinge(self, record):
        excess = 0.0
        for key, centre in self.truth["statistics"].items():
            value = record["statistics"].get(key)
            if value is None or not math.isfinite(value):
                continue
            z = _compressed((float(value) - centre) / self.scales[key])
            excess += max(0.0, abs(z) - TARGETS[key]) ** 2
        for m, zs in record["mutant_z"].items():
            for s, z in zs.items():
                if z is not None and math.isfinite(z):
                    excess += max(0.0, abs(_compressed(z)) - TARGETS[f"{m}/{s}"]) ** 2
        return HINGE_WEIGHT * excess

    def __call__(self, record):
        return super().__call__(record) + self.hinge(record)


def main():
    import cma

    OUT.mkdir(parents=True, exist_ok=True)
    state_path, record_path = OUT / "state.pkl", OUT.parent / "hinge2.json"
    if state_path.exists():
        saved = pickle.loads(state_path.read_bytes())
    else:
        z0 = json.loads(START.read_text())["z"]
        prior = pickle.loads(SEARCH_STATE.read_bytes())["es"]
        sigma0, stds = float(prior.sigma), np.asarray(prior.stds, dtype=float)
        options = {"CMA_diagonal": True, "CMA_stds": (stds / sigma0).tolist(), "bounds": [0.0, 1.0], "popsize": POPSIZE, "seed": CMA_SEED, "verbose": -9}
        saved = {"z0": z0, "sigma0": sigma0, "stds0": stds.tolist(), "targets": TARGETS, "es": cma.CMAEvolutionStrategy(z0, sigma0, options),
                 "medians": [], "best": [], "hinge_best": [], "readouts": {}, "verdict": None}
    objective = NoiseHingeObjective()
    t0 = time.time()
    with ProcessPoolExecutor(max_workers=WORKERS, initializer=nr._worker_setup) as pool:
        if not saved["readouts"]:
            saved["readouts"][0] = P.readout(pool, saved["z0"], READ)
            P._print("hinge2", 0, saved["readouts"][0])
        es = saved["es"]
        while len(saved["medians"]) < CAP:
            g = len(saved["medians"]) + 1
            points = es.ask()
            records = list(pool.map(simulate, [POPSIZE * (g - 1) + m for m in range(POPSIZE)], [list(map(float, x)) for x in points]))
            scores = [objective(r) for r in records]
            hinges = [objective.hinge(r) for r in records]
            es.tell(points, scores)
            saved["medians"].append(float(np.median(scores))); saved["best"].append(float(np.min(scores)))
            saved["hinge_best"].append(float(hinges[int(np.argmin(scores))]))
            print(f"hinge2 gen {g}: median {saved['medians'][-1]:.2f} best {saved['best'][-1]:.2f} (hinge {saved['hinge_best'][-1]:.2f}) sigma {es.sigma:.3f} {time.time() - t0:.0f} s", flush=True)
            if g % 10 == 0:
                saved["readouts"][g] = P.readout(pool, np.clip(es.result.xfavorite, 0, 1).tolist(), READ)
                P._print("hinge2", g, saved["readouts"][g])
            state_path.write_bytes(pickle.dumps(saved))
        z = np.clip(es.result.xfavorite, 0, 1).tolist()
        wilds = dict(pool.map(_wild, [(g, z) for g in range(5)]))
        for g, m, ok in pool.map(_mutant, [(g, m, z, wilds[g]["small_lawn"]) for g in wilds for m in p.MUTANTS]):
            wilds[g].setdefault("mutants", {})[m] = ok
        wt = sum(bool(w["wild_type_passes"]) for w in wilds.values())
        mut = {m: sum(w["mutants"][m] for w in wilds.values()) for m in p.MUTANTS}
        saved["grade"] = {str(g): {k: v for k, v in w.items() if k != "small_lawn"} for g, w in wilds.items()}
        saved["verdict"] = f"wild type {wt}/5, mutants {mut}: {'passes' if wt >= 3 and mut['tph-1'] >= 3 and mut['tax-4'] >= 3 else 'fails'}"
    saved["z"] = z
    state_path.write_bytes(pickle.dumps(saved))
    record_path.write_text(json.dumps(nr._finite({"description": __doc__, **{k: v for k, v in saved.items() if k != "es"}}), indent=1) + "\n")
    print(f"hinge2 done: {saved['verdict']}", flush=True)


if __name__ == "__main__":
    main()
