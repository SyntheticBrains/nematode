"""Is tph-1 reachable from the stage-3 search by the ser-5 route alone? (description; discarded seeds)

`serotonin_route.json` found serotonin's pool about 8x lower than in Logbook 129's answer and its
ser-5 couple about 30x weaker. Here the stage-3 search mean is graded as is and with ser-5's
strength raised to Logbook 129's (1.44), each on 200 animals of Scheer's small lawn as wild type
and as tph-1 (854,000,000 + i, the same animals for both), read by the probe's mutant reading.
"""
import json, pickle, sys
from concurrent.futures import ProcessPoolExecutor
import numpy as np
sys.path.insert(0, "scripts/analysis"); sys.path.insert(0, "build/c2-change-b")
import benchmark_grade as bg, c2_encoder_probe as p, npe_recovery as nr, probes as P

SEEDS = [854_000_000 + i for i in range(200)]


def job(item):
    z, mutant = item
    bands = bg.load_bands_v4()
    pops, _ = bg.simulate_v2(p.ARM, P._controller(z, mutant), SEEDS, bands, assays=("small_lawn",))
    return bg.family_values("small_lawn", pops["small_lawn"], bg.classifier(bands))


if __name__ == "__main__":
    s = pickle.load(open(sys.argv[1], "rb"))
    z = np.clip(s["es"].result.xfavorite, 0, 1)
    names = [e.name for e in P.entries()]
    j = next(i for i, n in enumerate(names) if n.startswith("modulator.fixed.") and n.endswith(".ser-5"))
    nudged = z.copy(); nudged[j] = P.entries()[j].unit(1.44)
    variants = {"as_is": z.tolist(), "ser5_at_1.44": nudged.tolist()}
    with ProcessPoolExecutor(max_workers=4, initializer=nr._worker_setup) as pool:
        values = list(pool.map(job, [(v, m) for v in variants.values() for m in (None, "tph-1")]))
    bands = bg.load_bands_v4(); out = {"entry": names[j], "ser5_unit_before": float(z[j]), "ser5_unit_after": float(nudged[j])}
    for i, name in enumerate(variants):
        wild, mut = values[2 * i], values[2 * i + 1]
        r = p.mutant_reading("tph-1", wild, mut, bands, 200)
        out[name] = {"passes": bool(r["passes"]), "statistics": {k: {"delta": v["delta"], "z": v["z"]} for k, v in r["statistics"].items()},
                     "wild_roaming_fraction": wild.get("roaming_fraction")}
        print(name, json.dumps(out[name]), flush=True)
    open("build/c2-change-b/serotonin_nudge.json", "w").write(json.dumps({"description": __doc__, **nr._finite(out)}, indent=1) + "\n")
