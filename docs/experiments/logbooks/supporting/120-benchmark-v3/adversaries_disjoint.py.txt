"""B v3's adversaries re-read as description on a seed block disjoint from the oracle's fit.

The registered adversary populations (472,000,000 + 200·k + i, k < 20) share animal seeds with the
oracle fit's restart 1 (472,000,000 to 472,319,999). This re-reads the four adversaries on
474,000,000 + 200·k + i, k = 5·i + g, which no other scheme uses. Description only: the registered
reading stands.
"""

import json
import sys
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

sys.path.insert(0, "scripts/analysis")
import benchmark_grade as bg
import benchmark_v3_controls as v3c
import npe_recovery as nr
from quantumnematode.benchmark import measurement as mm

BASE = 474_000_000
OUT = Path("docs/experiments/logbooks/supporting/120-benchmark-v3/adversaries_disjoint.json")


def population(item):
    name, params, k = item
    graded = v3c._mechanistic_graded(params, v3c.population_seeds(BASE, k), bg.load_bands_v3())
    summary = v3c._summary(graded)
    summary["caught"] = any(not r["passes"] for r in summary["new"].values())
    return {"name": name, "k": k, **summary}


def main():
    adversaries = list(v3c.adversary_parameters().items())
    jobs = [(n, p, v3c.GRADE_POPULATIONS * i + g) for i, (n, p) in enumerate(adversaries) for g in range(v3c.GRADE_POPULATIONS)]
    with ProcessPoolExecutor(16, initializer=nr._worker_setup) as pool:
        rows = list(pool.map(population, jobs))
    record = {
        "description": "Description: B v3's adversaries re-read on 474 M, disjoint from the oracle fit's seeds; the registered reading (472 M) stands.",
        "seed_base": BASE,
        "bands_sha256": mm.file_sha256(bg.BANDS_V3_PATH),
        "adversaries": v3c.adversary_verdicts(rows),
    }
    OUT.write_text(json.dumps(bg._finite(record), indent=2, allow_nan=False) + "\n")
    print({n: a["caught_in"] for n, a in record["adversaries"].items()})


if __name__ == "__main__":
    main()
