"""Does fitting hold CEP's route? A short continuation from the hand-nudged answer (description, discarded).

`grade_nudged.json`: the continuation's answer with CEP's food-contact gain raised by 0.8 passes the
wild type in 3 of 5 populations. Here sep-CMA continues from that answer at sigma0 0.02, with
`continuation.py`'s model, bounds and objective (the six families plus the three mutants at half the
objective, slow speed noise, both guards), at population 48, 200 animals, 25 generations on 16
workers. Readouts every 5 generations on 200 animals (801,000,000 + i) record the on-lawn onset
rate's z and the search mean's CEP food-contact gain.

The reading, fixed before the run: fitting holds the route if at generation 25 the onset rate is
inside its band (|z| <= 3.078) and CEP's gain has kept at least half the nudge (at least its fitted
value + 0.4); otherwise it drifts, and C.2's change needs a design element for the on-lawn onset rate.

Seeds: fit animals 800,000,000 + 200·k + i; CMA 9489. Run from the repository root.
"""
import json, sys, time
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path
import numpy as np
sys.path.insert(0, "scripts/analysis"); sys.path.insert(0, "build/c2-run-mode")
import continuation as C, fit_mutants as F, leave_one_out as L, npe_recovery as nr, overnight as O, screens as s, speednoise as sn

L.BASE = 800_000_000
s.READ_SEEDS = [801_000_000 + i for i in range(200)]
POPSIZE, SIGMA0, CAP, EVERY, CMA_SEED = 48, 0.02, 25, 5, 9489
CEP = next(i for i, e in enumerate(sn.ENTRIES) if e.name == "sensory.CEP.contact")
FITTED = json.loads(Path("build/c2-run-mode/continuation.json").read_text())["answer"]


def cep(z):
    return -2.0 + 4.0 * float(z[CEP])  # the graded box: value = -2 + 4u


def main():
    import cma
    x0 = json.loads(Path("build/c2-run-mode/grade_nudged.json").read_text())["answer"]
    lower, upper = F.bounds(len(x0) - 1)
    lower, upper = lower + [0.0], upper + [1.0]
    es = cma.CMAEvolutionStrategy(x0, SIGMA0, {"CMA_diagonal": True, "bounds": [lower, upper], "popsize": POPSIZE, "seed": CMA_SEED, "verbose": -9})
    objective, history, readouts, t0 = O.Objective(), {"medians": [], "best": [], "cep": []}, {}, time.time()
    with ProcessPoolExecutor(max_workers=16, initializer=nr._worker_setup) as pool:
        readouts[0] = s.readout(pool, x0)
        s._print("S", 0, readouts[0])
        for g in range(1, CAP + 1):
            points = es.ask()
            records = list(pool.map(L.simulate, [POPSIZE * (g - 1) + m for m in range(POPSIZE)], [list(map(float, x)) for x in points]))
            scores = [objective(r) for r in records]
            es.tell(points, scores)
            mean = np.clip(es.result.xfavorite, lower, upper).tolist()
            history["medians"].append(float(np.median(scores))); history["best"].append(float(np.min(scores))); history["cep"].append(cep(mean))
            print(f"S gen {g}: median {history['medians'][-1]:.2f} best {history['best'][-1]:.2f} CEP {history['cep'][-1]:.3f} {time.time() - t0:.0f} s", flush=True)
            if g % EVERY == 0:
                readouts[g] = s.readout(pool, mean)
                s._print("S", g, readouts[g])
    last = readouts[CAP]
    onset = next(v for k, v in last["fails"].items() if k == "activity/onset_rate_per_min") if "activity/onset_rate_per_min" in last["fails"] else None
    holds = onset is None and history["cep"][-1] >= cep(FITTED) + 0.4
    record = {"description": __doc__, **history, "readouts": readouts, "cep_fitted": cep(FITTED), "cep_nudged": cep(x0), "answer": mean, "holds": holds}
    Path("build/c2-run-mode/stability.json").write_text(json.dumps(nr._finite(record), indent=1) + "\n")
    print(f"S done: CEP fitted {cep(FITTED):.3f}, nudged {cep(x0):.3f}, final {history['cep'][-1]:.3f}; onset failing {onset}; holds {holds}", flush=True)


if __name__ == "__main__":
    main()
